   114→/* ------------ TTL burn scheduling (module memory) ------------ */
   115→const burnTimers = new Map<string, ReturnType<typeof setTimeout>>();
   116→
   117→/* ------------ typing whispers (transient by design) ------------ */
   118→const TYPING_TTL_MS = 3500;
   119→const TYPING_EMIT_THROTTLE_MS = 2500;
   120→const typingLastEmit = new Map<string, number>();
   121→const typingPruneTimers = new Map<string, ReturnType<typeof setTimeout>>();
   122→
   123→function pruneTypingLater(roomId: string) {
   124→  const existing = typingPruneTimers.get(roomId);
   125→  if (existing) clearTimeout(existing);
   126→  const t = setTimeout(() => {
   127→    typingPruneTimers.delete(roomId);
   128→    useApp.setState((s) => {
   129→      const fresh = (s.typing[roomId] ?? []).filter(
   130→        (x) => Date.now() - x.at < TYPING_TTL_MS,
   131→      );
   132→      return { typing: { ...s.typing, [roomId]: fresh } };
   133→    });
   134→  }, TYPING_TTL_MS + 100);
   135→  typingPruneTimers.set(roomId, t);
   136→}
   137→
   138→function clearTyping(roomId: string, memberId: string) {
   139→  useApp.setState((s) => {
   140→    const list = s.typing[roomId] ?? [];
   141→    if (!list.some((t) => t.memberId === memberId)) return {};
   142→    return {
   143→      typing: {
   144→        ...s.typing,
   145→        [roomId]: list.filter((t) => t.memberId !== memberId),
   146→      },
   147→    };
   148→  });
   149→}
   150→
   151→function scheduleBurn(
   152→  roomId: string,
   153→  messageId: string,
   154→  expiresAt: number,
   155→  onBurn: (roomId: string, messageId: string) => void,
   156→) {
   157→  if (burnTimers.has(messageId)) return;
   158→  const delay = Math.max(0, expiresAt - Date.now());
   159→  const t = setTimeout(() => {
   160→    burnTimers.delete(messageId);
   161→    onBurn(roomId, messageId);
   162→  }, delay);
   163→  burnTimers.set(messageId, t);
   164→}
   165→
   166→function cancelBurns(roomId: string, messages: MessageView[]) {
   167→  for (const m of messages) {
   168→    if (m.id) {
   169→      const t = burnTimers.get(m.id);
   170→      if (t) {
   171→        clearTimeout(t);
   172→        burnTimers.delete(m.id);
   173→      }
   174→    }
   175→  }
   176→  void roomId;
   177→}

   182→  ready: false,
   183→  device: null,
   184→  screen: "landing",

   186→  inviteCode: null,
   187→

   197→  /* -------------------------------------------------- init ---- */
   198→
   199→  init: async () => {
   200→    if (get().ready) return;
   201→    const device = await loadDeviceIdentity();
   202→    // Burned rooms showed their ash last session; now they are gone.
   203→    const cards = sweepBurnedRooms();
   204→    set({ device, roomCards: cards, ready: true });
   205→
   206→    get().syncHash();
   207→    window.addEventListener("hashchange", () => get().syncHash());
   208→
   209→    wireRelay(set, get);
   210→  },
   211→
   212→  syncHash: () => {
   213→    const hash = window.location.hash.replace(/^#\/?/, "");
   214→    if (hash.startsWith("join")) {
   215→      const code = hash.split("/")[1] ?? "";
   216→      set({ screen: "invite", inviteCode: code || null, activeRoomId: null });
   217→      return;
   218→    }
   219→    if (hash.startsWith("new")) {
   220→      set({ screen: "landing", activeRoomId: null });
   221→      return;
   222→    }
   223→    if (hash.startsWith("rooms")) {
   224→      set({ screen: "rooms", activeRoomId: null });
   225→      return;
   226→    }
   227→    if (hash.startsWith("r/")) {
   228→      const roomId = hash.slice(2);
   229→      const session = getSession(roomId);
   230→      set({ screen: "chat", activeRoomId: roomId });
   231→      if (session) {
   232→        patchRoomCard(roomId, { unread: false });
   233→        set({ roomCards: loadRoomCards() });
   234→      }
   235→      // No keys in memory — the room is locked. A first-class state,
   236→      // never an error.
   237→      return;
   238→    }
   239→    // Root: the desk if you have rooms, the landing if you don't.
   240→    if (get().roomCards.length > 0) {
   241→      set({ screen: "rooms", activeRoomId: null });
   242→    } else {
   243→      set({ screen: "landing", activeRoomId: null });
   244→    }
   245→  },
   246→
   247→  navigate: (screen, roomId = null) => {
   248→    const hash =
   249→      screen === "rooms"
   250→        ? "#/rooms"
   251→        : screen === "chat" && roomId
   252→          ? `#/r/${roomId}`
   253→          : screen === "invite"
   254→            ? "#/join"
   255→            : screen === "landing"
   256→              ? "#/new"
   257→              : "#/";
   258→    if (window.location.hash !== hash) {
   259→      window.location.hash = hash;
   260→    } else {
   261→      get().syncHash();
   262→    }
   263→  },
   264→
   265→  /* ----------------------------------------------- create ---- */
   266→
   267→  createRoom: async (localName, password) => {
   268→    const device = get().device;
   269→    if (!device) return { ok: false, reason: "error" };
   270→
   271→    const res = await fetch("/api/rooms", { method: "POST" });
   272→    if (!res.ok) return { ok: false, reason: "error" };
   273→    const { roomId, creatorToken } = await res.json();
   274→    saveCreatorToken(roomId, creatorToken);
   275→
   276→    set({ sealing: { roomId, label: "Sealing the room" } });
   277→    const started = Date.now();
   278→
   279→    const { deriveRoomKey } = await import("@/lib/crypto");
   280→    const key = await deriveRoomKey(roomId, 1, password);
   281→    const verifier = await makeVerifier(key);
   282→    await Promise.all([
   283→      fetch(`/api/rooms/${roomId}/verifier`, {
   284→        method: "PUT",
   285→        headers: { "content-type": "application/json" },
   286→        body: JSON.stringify({ verifier }),
   287→      }),
   288→      new Promise((r) => setTimeout(r, MIN_SEAL_MS - (Date.now() - started))),
   289→    ]);
   290→
   291→    const alias = aliasFromFingerprint(device.fingerprintHex);
   292→    const colorIdx = inkFromFingerprint(device.fingerprintHex);
   293→    const joinRes = await fetch(`/api/rooms/${roomId}/members`, {
   294→      method: "POST",
   295→      headers: { "content-type": "application/json" },
   296→      body: JSON.stringify({ pubkey: device.pubJwk, alias, colorIdx }),
   297→    });
   298→    if (!joinRes.ok) {
   299→      set({ sealing: null });
   300→      return { ok: false, reason: "error" };
   301→    }
   302→    const { memberId } = await joinRes.json();
   303→
   304→    const session: RoomSession = {
   305→      roomId,
   306→      memberId,
   307→      alias,
   308→      colorIdx,
   309→      epoch: 1,
   310→      password,
   311→      creatorToken,
   312→      defaultTtl: loadRoomSettings(roomId).defaultTtl,
   313→      keys: new Map([[1, key]]),
   314→      pending: new Map(),
   315→    };
   316→    setSession(session);
   317→    set({ sealing: null });
   318→    await get().enterRoom(session, localName || "New room", false);
   319→    return { ok: true };
   320→  },
   321→
   322→  /* ------------------------------------------------- join ---- */
   323→
   324→  joinRoom: async (code, password, localName) => {
   325→    const device = get().device;
   326→    if (!device) return { ok: false, reason: "error" };
   327→
   328→    const infoRes = await fetch(`/api/rooms/${encodeURIComponent(code)}`);
   329→    if (!infoRes.ok) {
   330→      const info = await infoRes.json().catch(() => ({}));
   331→      return {
   332→        ok: false,
   333→        reason: info?.burned ? "burned" : "not-found",
   334→      };
   335→    }
   336→    const info = await infoRes.json();
   337→    if (info.memberCount >= info.memberCap) {
   338→      return { ok: false, reason: "room-full" };
   339→    }
   340→
   341→    const { deriveRoomKey } = await import("@/lib/crypto");
   342→    set({ sealing: { roomId: code, label: "Sealing the room" } });
   343→    const started = Date.now();
   344→    const key = await deriveRoomKey(code, info.epoch, password);
   345→    const verified = info.verifier ? await checkVerifier(key, info.verifier) : true;
   346→    await new Promise((r) =>
   347→      setTimeout(r, Math.max(0, MIN_SEAL_MS - (Date.now() - started))),
   348→    );
   349→
   350→    if (!verified) {
   351→      set({ sealing: null });
   352→      return { ok: false, reason: "wrong-password" };
   353→    }
   354→
   355→    const alias = aliasFromFingerprint(device.fingerprintHex);
   356→    const colorIdx = inkFromFingerprint(device.fingerprintHex);
   357→    const joinRes = await fetch(`/api/rooms/${code}/members`, {
   358→      method: "POST",
   359→      headers: { "content-type": "application/json" },
   360→      body: JSON.stringify({ pubkey: device.pubJwk, alias, colorIdx }),
   361→    });
   362→    if (!joinRes.ok) {
   363→      set({ sealing: null });
   364→      if (joinRes.status === 403) return { ok: false, reason: "room-full" };
   365→      return { ok: false, reason: "error" };
   366→    }
   367→    const { memberId, epoch, rejoined } = await joinRes.json();
   368→
   369→    const session: RoomSession = {
   370→      roomId: code,
   371→      memberId,
   372→      alias,
   373→      colorIdx,
   374→      epoch,
   375→      password,
   376→      creatorToken: loadCreatorToken(code),
   377→      defaultTtl: loadRoomSettings(code).defaultTtl,
   378→      keys: new Map([[epoch, key]]),
   379→      pending: new Map(),
   380→    };
   381→    setSession(session);
   382→    set({ sealing: null });
   383→    const name =
   384→      localName ||
   385→      get().roomCards.find((c) => c.roomId === code)?.localName ||
   386→      `Room ${code.slice(0, 4)}`;
   387→    await get().enterRoom(session, name, !!rejoined);
   388→    return { ok: true };
   389→  },
   390→
   391→  /* ----------------------------------------------- enter ----- */
   392→
   393→  enterRoom: async (session, localName, rejoined) => {
   394→    const now = Date.now();
   395→    upsertRoomCard({
   396→      roomId: session.roomId,
   397→      localName,
   398→      createdAt: now,
   399→      lastActivity: now,
   400→      lastMembers: 1,
   401→      unread: false,
   402→      burned: false,
   403→      locked: false,
   404→    });
   405→    set({ roomCards: loadRoomCards() });
   406→
   407→    // Fresh message memory — there is no backlog, ever.
   408→    set((s) => ({ messages: { ...s.messages, [session.roomId]: [] } }));
   409→
   410→    await get().refreshMembers(session.roomId);
   411→
   412→    const device = get().device;
   413→    const relay = getRelay();
   414→    relay.emit("room:join", {
   415→      roomId: session.roomId,
   416→      memberId: session.memberId,
   417→      alias: session.alias,
   418→      colorIdx: session.colorIdx,
   419→      pubkey: device?.pubJwk,
   420→      rejoined,
   421→    });
   422→    set({ relayOnline: relay.connected });
   423→
   424→    get().navigate("chat", session.roomId);
   425→  },
   426→
   427→  refreshMembers: async (roomId) => {
   428→    try {
   429→      const res = await fetch(`/api/rooms/${roomId}/members`);
   430→      if (!res.ok) return;
   431→      const { members } = await res.json();
   432→      set((s) => ({ members: { ...s.members, [roomId]: members } }));
   433→    } catch {
   434→      /* offline — the registry stays as-is */
   435→    }
   436→  },
   437→
   438→  /* ------------------------------------------------- send ---- */
   439→
   440→  sendMessage: async (text, file, ttlOverride) => {
   441→    const viewOnce = !!file?.viewOnce;
   442→    const { activeRoomId, device } = get();
   443→    if (!activeRoomId || !device) return;
   444→    const session = getSession(activeRoomId);
   445→    if (!session) return;
   446→    const trimmed = text.trim();
   447→    if (!trimmed && !file) return;
   448→
   449→    const id = crypto.randomUUID();
   450→    const ts = Date.now();
   451→    const kind = file ? "file" : "text";
   452→    let fileSha: string | undefined;
   453→    let fullFile: FilePayload | undefined;
   454→    if (file) {
   455→      fileSha = await sha256HexOfBytes(fromB64(file.dataB64));
   456→      fullFile = { ...file, sha: fileSha };
   457→    }
   458→
   459→    const canonical = canonicalFor({
   460→      roomId: session.roomId,
   461→      epoch: session.epoch,
   462→      senderId: session.memberId,
   463→      ts,
   464→      kind,
   465→      text: trimmed,
   466→      fileSha,
   467→    });
   468→    const sig = await signCanonical(device.privJwk, canonical);
   469→    const payload: Payload = {
   470→      text: trimmed,
   471→      ts,
   472→      senderId: session.memberId,
   473→      sig,
   474→      kind,
   475→      file: fullFile,
   476→    };
   477→
   478→    const key = await getEpochKey(session, session.epoch);
   479→    const { iv, ct } = await encryptJson(key, payload);
   480→
   481→    const ttlSec = (ttlOverride !== undefined ? ttlOverride : session.defaultTtl) || undefined;
   482→    const envelope: WireEnvelope = {
   483→      id,
   484→      roomId: session.roomId,
   485→      epoch: session.epoch,
   486→      senderId: session.memberId,
   487→      ts,
   488→      iv,
   489→      ct,
   490→      ttlSec,
   491→      viewOnce: file && viewOnce ? true : undefined,
   492→      kind,
   493→    };
   494→
   495→    const view: MessageView = {
   496→      id,
   497→      kind,
   498→      senderId: session.memberId,
   499→      senderAlias: session.alias,
   500→      senderColor: session.colorIdx,
   501→      self: true,
   502→      status: "sending",
   503→      ts,
   504→      text: trimmed,
   505→      file: file
   506→        ? { name: file.name, mime: file.mime, size: file.size, dataB64: file.dataB64 }
   507→        : undefined,
   508→      ttlSec,
   509→      expiresAt: ttlSec ? ts + ttlSec * 1000 : undefined,
   510→      viewOnce: file && viewOnce ? true : undefined,
   511→    };
   512→
   513→    set((s) => ({
   514→      messages: {
   515→        ...s.messages,
   516→        [session.roomId]: [...(s.messages[session.roomId] ?? []), view],
   517→      },
   518→    }));
   519→    patchRoomCard(session.roomId, { lastActivity: ts });
   520→    set({ roomCards: loadRoomCards() });
   521→
   522→    if (ttlSec) {
   523→      scheduleBurn(session.roomId, id, ts + ttlSec * 1000, onMessageBurn);
   524→    }
   525→
   526→    getRelay().emit("message:send", { roomId: session.roomId, envelope });
   527→  },
   528→
   529→  /* --------------------------------------------- typing ------- */
   530→
   531→  emitTyping: (roomId) => {
   532→    const session = getSession(roomId);
   533→    if (!session) return;
   534→    const now = Date.now();
   535→    const last = typingLastEmit.get(roomId) ?? 0;
   536→    if (now - last < TYPING_EMIT_THROTTLE_MS) return;
   537→    typingLastEmit.set(roomId, now);
   538→    getRelay().emit("member:typing", {
   539→      roomId,
   540→      memberId: session.memberId,
   541→      alias: session.alias,
   542→    });
   543→  },
   544→
   545→  /* --------------------------------------------- view-once --- */
   546→
   547→  spendViewOnce: (roomId, messageId) => {
   548→    set((s) => {
   549→      const list = s.messages[roomId] ?? [];
   550→      return {
   551→        messages: {
   552→          ...s.messages,
   553→          [roomId]: list.map((m) => (m.id === messageId ? { ...m, spent: true } : m)),
   554→        },
   555→      };
   556→    });
   557→    getRelay().emit("message:spent", { roomId, messageId });
   558→  },
   559→
   560→  /* --------------------------------------------- burn-one ---- */
   561→
   562→  /** Retire one of your own letters ahead of its clock. The announce
   563→   *  is signed, so only the author can do this — nobody else can
   564→   *  burn a message they did not write. */
   565→  burnMessage: async (roomId, messageId) => {
   566→    const { activeRoomId, device } = get();
   567→    if (!device) return;
   568→    const target = roomId ?? activeRoomId;
   569→    const session = getSession(target);
   570→    const msg = (get().messages[target] ?? []).find((m) => m.id === messageId);
   571→    if (!session || !msg || !msg.self || msg.kind === "system") return;
   572→
   573→    const sig = await signCanonical(
   574→      device.privJwk,
   575→      burnCanonical(target, session.memberId, messageId),
   576→    );
   577→    getRelay().emit("message:burn", {
   578→      roomId: target,
   579→      messageId,
   580→      senderId: session.memberId,
   581→      sig,
   582→    });
   583→
   584→    const t = burnTimers.get(messageId);
   585→    if (t) {
   586→      clearTimeout(t);
   587→      burnTimers.delete(messageId);
   588→    }
   589→    onMessageBurn(target, messageId);
   590→  },
   591→
   592→  /* ------------------------------------------------- burn ---- */
   593→
   594→  burnRoom: async (roomId) => {
   595→    const session = getSession(roomId);
   596→    if (!session?.creatorToken) return;
   597→    try {
   598→      await fetch(`/api/rooms/${roomId}/burn`, {
   599→        method: "POST",
   600→        headers: { "content-type": "application/json" },
   601→        body: JSON.stringify({ creatorToken: session.creatorToken }),
   602→      });
   603→    } catch {
   604→      /* even if this fails, members burn on the announce */
   605→    }
   606→    getRelay().emit("room:burn", { roomId });
   607→    set({ burn: { roomId } });
   608→  },
   609→
   610→  finishRoomBurn: (roomId) => {
   611→    const messages = get().messages[roomId] ?? [];
   612→    cancelBurns(roomId, messages);
   613→    dropSession(roomId);
   614→    patchRoomCard(roomId, { burned: true, unread: false, lastActivity: Date.now() });
   615→    set((s) => ({
   616→      burn: null,
   617→      roomCards: loadRoomCards(),
   618→      activeRoomId: null,
   619→      messages: { ...s.messages, [roomId]: [] },
   620→    }));
   621→    get().navigate("rooms");
   622→  },
   623→
   624→  /* ------------------------------------------------- leave --- */
   625→
   626→  leaveRoom: async (roomId) => {
   627→    const session = getSession(roomId);
   628→    if (session) {
   629→      try {
   630→        await fetch(`/api/rooms/${roomId}/leave`, {
   631→          method: "POST",
   632→          headers: { "content-type": "application/json" },
   633→          body: JSON.stringify({ memberId: session.memberId }),
   634→        });
   635→      } catch {
   636→        /* best effort — the epoch bump happens server-side anyway */
   637→      }
   638→      getRelay().emit("member:leave", {
   639→        roomId,
   640→        memberId: session.memberId,
   641→        alias: session.alias,
   642→      });
   643→    }
   644→    const messages = get().messages[roomId] ?? [];
   645→    cancelBurns(roomId, messages);
   646→    dropSession(roomId);
   647→    removeRoomCard(roomId);
   648→    set((s) => ({
   649→      roomCards: loadRoomCards(),
   650→      activeRoomId: null,
   651→      messages: { ...s.messages, [roomId]: [] },
   652→    }));
   653→    get().navigate("rooms");
   654→  },
   655→
   656→  /* ---------------------------------------------- settings --- */
   657→
   658→  renameRoom: (roomId, name) => {
   659→    patchRoomCard(roomId, { localName: name });
   660→    set({ roomCards: loadRoomCards() });
   661→  },
   662→
   663→  setDefaultTtl: (roomId, ttl) => {
   664→    const session = getSession(roomId);
   665→    if (session) session.defaultTtl = ttl;
   666→    saveRoomSettings(roomId, { defaultTtl: ttl });
   667→  },
   668→
   669→  markVerified: (roomId, memberId, verified) => {
   670→    const current = loadVerified(roomId);
   671→    const next = verified
   672→      ? Array.from(new Set([...current, memberId]))
   673→      : current.filter((id) => id !== memberId);
   674→    saveVerified(roomId, next);
   675→    set((s) => ({ members: { ...s.members } })); // trigger re-render
   676→  },
   677→
   678→  isVerified: (roomId, memberId) => loadVerified(roomId).includes(memberId),
   679→}));

   685→function onMessageBurn(roomId: string, messageId: string) {
   686→  // Phase 1: burning (CSS animation 600ms)
   687→  useApp.setState((s) => ({
   688→    messages: {
   689→      ...s.messages,
   690→      [roomId]: (s.messages[roomId] ?? []).map((m) =>
   691→        m.id === messageId ? { ...m, status: "burning" } : m,
   692→      ),
   693→    },
   694→  }));
   695→  // Phase 2: gone.
   696→  setTimeout(() => {
   697→    useApp.setState((s) => ({
   698→      messages: {
   699→        ...s.messages,
   700→        [roomId]: (s.messages[roomId] ?? []).filter((m) => m.id !== messageId),
   701→      },
   702→    }));
   703→  }, 620);
   704→}

   706→function addSystemLine(roomId: string, text: string) {
   707→  const line: MessageView = {
   708→    id: crypto.randomUUID(),
   709→    kind: "system",
   710→    status: "sent",
   711→    ts: Date.now(),
   712→    text,
   713→  };
   714→  useApp.setState((s) => ({
   715→    messages: {
   716→      ...s.messages,
   717→      [roomId]: [...(s.messages[roomId] ?? []), line],
   718→    },
   719→  }));
   720→}

   722→function patchView(roomId: string, messageId: string, patch: Partial<MessageView>) {
   723→  useApp.setState((s) => ({
   724→    messages: {
   725→      ...s.messages,
   726→      [roomId]: (s.messages[roomId] ?? []).map((m) =>
   727→        m.id === messageId ? { ...m, ...patch } : m,
   728→      ),
   729→    },
   730→  }));
   731→}

   738→function wireRelay(
   739→  set: (
   740→    partial:
   741→      | Partial<AppState>
   742→      | ((state: AppState) => Partial<AppState>),
   743→  ) => void,
   744→  get: () => AppState,
   745→) {
   746→  const relay = getRelay();
   747→
   748→  relay.on("connect", () => {
   749→    set({ relayOnline: true });
   750→    const device = useApp.getState().device;
   751→    for (const roomId of sessionRoomIds()) {
   752→      const session = getSession(roomId);
   753→      if (session) {
   754→        relay.emit("room:join", {
   755→          roomId,
   756→          memberId: session.memberId,
   757→          alias: session.alias,
   758→          colorIdx: session.colorIdx,
   759→          pubkey: device?.pubJwk,
   760→          rejoined: true,
   761→        });
   762→      }
   763→    }
   764→  });
   765→
   766→  relay.on("disconnect", () => set({ relayOnline: false }));
   767→
   768→  relay.on("room:state", ({ roomId, members }: { roomId: string; members: MemberPublic[] }) => {
   769→    // merge live presence into the registry
   770→    const registry = useApp.getState().members[roomId] ?? [];
   771→    const byId = new Map(registry.map((m) => [m.memberId, { ...m }]));
   772→    for (const live of members) {
   773→      const existing = byId.get(live.memberId);
   774→      byId.set(live.memberId, {
   775→        ...live,
   776→        ...(existing?.pubkey ? { pubkey: existing.pubkey } : {}),
   777→      });
   778→    }
   779→    set((s) => ({ members: { ...s.members, [roomId]: Array.from(byId.values()) } }));
   780→  });
   781→
   782→  relay.on(
   783→    "member:joined",
   784→    ({ roomId, member, rejoined }: { roomId: string; member: MemberPublic; rejoined?: boolean }) => {
   785→      const session = getSession(roomId);
   786→      if (!session) return;
   787→      set((s) => {
   788→        const registry = s.members[roomId] ?? [];
   789→        if (registry.some((m) => m.memberId === member.memberId)) {
   790→          return {
   791→            members: {
   792→              ...s.members,
   793→              [roomId]: registry.map((m) =>
   794→                m.memberId === member.memberId ? { ...m, connected: true } : m,
   795→              ),
   796→            },
   797→          };
   798→        }
   799→        return {
   800→          members: { ...s.members, [roomId]: [...registry, { ...member, connected: true }] },
   801→        };
   802→      });
   803→      if (member.memberId !== session.memberId) {
   804→        addSystemLine(roomId, rejoined ? `${member.alias} returned.` : `${member.alias} joined.`);
   805→        touchCard(roomId, get().activeRoomId !== roomId);
   806→      }
   807→    },
   808→  );
   809→
   810→  relay.on(
   811→    "member:presence",
   812→    ({ roomId, memberId, connected }: { roomId: string; memberId: string; connected: boolean }) => {
   813→      set((s) => ({
   814→        members: {
   815→          ...s.members,
   816→          [roomId]: (s.members[roomId] ?? []).map((m) =>
   817→            m.memberId === memberId ? { ...m, connected } : m,
   818→          ),
   819→        },
   820→      }));
   821→    },
   822→  );
   823→
   824→  relay.on(
   825→    "member:left",
   826→    async ({ roomId, alias }: { roomId: string; memberId: string; alias: string }) => {
   827→      const session = getSession(roomId);
   828→      if (!session) return;
   829→      addSystemLine(roomId, `${alias} left. The room key was rotated.`);
   830→      // Rotate: fetch the new epoch and re-derive quietly.
   831→      set((s) => ({ resealing: { ...s.resealing, [roomId]: true } }));
   832→      try {
   833→        const res = await fetch(`/api/rooms/${roomId}`);
   834→        if (res.ok) {
   835→          const info = await res.json();
   836→          session.epoch = info.epoch;
   837→          await getEpochKey(session, info.epoch);
   838→        }
   839→      } finally {
   840→        set((s) => {
   841→          const next = { ...s.resealing };
   842→          delete next[roomId];
   843→          return { resealing: next };
   844→        });
   845→      }
   846→      get().refreshMembers(roomId);
   847→    },
   848→  );
   849→
   850→  relay.on("message:ack", ({ id, roomId }: { id: string; roomId?: string }) => {
   851→    const target = roomId ?? get().activeRoomId;
   852→    if (!target) return;
   853→    patchView(target, id, { status: "sent" });
   854→  });
   855→
   856→  relay.on("message:spent", ({ roomId, messageId }: { roomId: string; messageId: string }) => {
   857→    patchView(roomId, messageId, { spent: true });
   858→  });
   859→
   860→  relay.on(
   861→    "message:burn",
   862→    async ({
   863→      roomId,
   864→      messageId,
   865→      senderId,
   866→      sig,
   867→    }: {
   868→      roomId: string;
   869→      messageId: string;
   870→      senderId?: string;
   871→      sig?: string;
   872→    }) => {
   873→      const session = getSession(roomId);
   874→      if (!session || !senderId) return;
   875→      const target = (useApp.getState().messages[roomId] ?? []).find(
   876→        (m) => m.id === messageId,
   877→      );
   878→      if (!target || target.kind === "system") return;
   879→
   880→      const registry = useApp.getState().members[roomId] ?? [];
   881→      const author = registry.find((m) => m.memberId === senderId);
   882→      const ok =
   883→        !!author?.pubkey &&
   884→        !!sig &&
   885→        target.senderId === senderId &&
   886→        (await verifyCanonical(author.pubkey, burnCanonical(roomId, senderId, messageId), sig));
   887→      if (!ok) {
   888→        addSystemLine(
   889→          roomId,
   890→          `A burn request claiming to be from ${author?.alias ?? "an unknown member"} was rejected — signature invalid.`,
   891→        );
   892→        return;
   893→      }
   894→      const t = burnTimers.get(messageId);
   895→      if (t) {
   896→        clearTimeout(t);
   897→        burnTimers.delete(messageId);
   898→      }
   899→      onMessageBurn(roomId, messageId);
   900→    },
   901→  );
   902→
   903→  relay.on("room:burned", ({ roomId }: { roomId: string }) => {
   904→    const session = getSession(roomId);
   905→    if (!session) return;
   906→    set({ burn: { roomId } });
   907→  });
   908→
   909→  relay.on(
   910→    "message:new",
   911→    async ({ roomId, envelope }: { roomId: string; envelope: WireEnvelope }) => {
   912→      const session = getSession(roomId);
   913→      if (!session || !envelope) return;
   914→
   915→      // Might need a key for a newer epoch (quiet re-seal)
   916→      set((s) => ({ resealing: { ...s.resealing, [roomId]: true } }));
   917→      let payload: Payload | null = null;
   918→      try {
   919→        const key = await getEpochKey(session, envelope.epoch);
   920→        payload = await decryptJson<Payload>(key, envelope.iv, envelope.ct);
   921→      } catch {
   922→        payload = null;
   923→      } finally {
   924→        set((s) => {
   925→          const next = { ...s.resealing };
   926→          delete next[roomId];
   927→          return { resealing: next };
   928→        });
   929→      }
   930→
   931→      const registry = get().members[roomId] ?? [];
   932→      const sender = registry.find((m) => m.memberId === envelope.senderId);
   933→
   934→      if (!payload || !sender) {
   935→        // Tampered, undecryptable, or from an unknown member: rejected.
   936→        addSystemLine(
   937→          roomId,
   938→          `A message claiming to be from ${sender?.alias ?? "an unknown member"} was rejected — signature invalid.`,
   939→        );
   940→        touchCard(roomId, get().activeRoomId !== roomId);
   941→        return;
   942→      }
   943→
   944→      const canonical = canonicalFor({
   945→        roomId,
   946→        epoch: envelope.epoch,
   947→        senderId: envelope.senderId,
   948→        ts: payload.ts,
   949→        kind: envelope.kind,
   950→        text: payload.text,
   951→        fileSha: payload.file?.sha,
   952→      });
   953→      const ok = await verifyCanonical(sender.pubkey, canonical, payload.sig);
   954→      if (!ok) {
   955→        addSystemLine(
   956→          roomId,
   957→          `A message claiming to be from ${sender.alias} was rejected — signature invalid.`,
   958→        );
   959→        touchCard(roomId, get().activeRoomId !== roomId);
   960→        return;
   961→      }
   962→
   963→      if (envelope.epoch > session.epoch) session.epoch = envelope.epoch;
   964→
   965→      const view: MessageView = {
   966→        id: envelope.id,
   967→        kind: envelope.kind,
   968→        senderId: sender.memberId,
   969→        senderAlias: sender.alias,
   970→        senderColor: sender.colorIdx,
   971→        self: sender.memberId === session.memberId,
   972→        status: "sent",
   973→        ts: payload.ts,
   974→        text: payload.text,
   975→        file: payload.file
   976→          ? {
   977→              name: payload.file.name,
   978→              mime: payload.file.mime,
   979→              size: payload.file.size,
   980→              dataB64: payload.file.dataB64,
   981→            }
   982→          : undefined,
   983→        ttlSec: envelope.ttlSec,
   984→        expiresAt: envelope.ttlSec ? payload.ts + envelope.ttlSec * 1000 : undefined,
   985→        viewOnce: envelope.viewOnce,
   986→      };
   987→      useApp.setState((s) => ({
   988→        messages: {
   989→          ...s.messages,
   990→          [roomId]: [...(s.messages[roomId] ?? []), view],
   991→        },
   992→      }));
   993→      // The message arrived — the whisper is over.
   994→      clearTyping(roomId, sender.memberId);
   995→      touchCard(roomId, get().activeRoomId !== roomId);
   996→      if (view.expiresAt) {
   997→        scheduleBurn(roomId, view.id, view.expiresAt, onMessageBurn);
   998→      }
   999→    },
  1000→  );
  1001→
  1002→  relay.on(
  1003→    "member:typing",
  1004→    ({ roomId, memberId, alias }: { roomId: string; memberId: string; alias: string }) => {
  1005→      const session = getSession(roomId);
  1006→      if (!session || memberId === session.memberId) return;
  1007→      useApp.setState((s) => {
  1008→        const list = (s.typing[roomId] ?? []).filter((t) => t.memberId !== memberId);
  1009→        return {
  1010→          typing: { ...s.typing, [roomId]: [...list, { memberId, alias, at: Date.now() }] },
  1011→        };
  1012→      });
  1013→      pruneTypingLater(roomId);
  1014→    },
  1015→  );
  1016→}
