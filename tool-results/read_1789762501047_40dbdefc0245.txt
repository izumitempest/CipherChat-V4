     1→// RoomCipher — the per-room security engine. One instance per joined
     2→// room, memory only. It owns:
     3→//
     4→//   the key ring         versioned room keys (v1 = password-derived
     5→//                        entry key; later versions = random keys
     6→//                        delivered over ECDH; old versions kept for a
     7→//                        short decrypt-only grace window)
     8→//   the member registry  frames only flow between current members —
     9→//                        a departed member is evicted here first
    10→//   the replay guard     counters / timestamps / frame ids
    11→//   the file assembler   fixed-count chunk frames back into bytes
    12→//
    13→// The UI store is a thin adapter around this class; every security
    14→// decision lives here, where the Task 19 tests can reach it.
    15→

   118→/** Hard ceiling for key versions. Offers claiming versions beyond this
   119→ *  are rejected: a hostile member could otherwise install kv = 2^53 and
   120→ *  permanently brick every future rotation (newKv would equal kv). */
   121→export const KVERSION_CAP = 1_000_000;
   122→
   123→interface InstalledKey {
   124→  key: CryptoKey;
   125→  raw: Uint8Array; // empty for the entry key (non-extractable)
   126→  installedAt: number;
   127→}

   137→export class RoomCipher {
   138→  readonly roomId: string;
   139→  readonly selfId: string;
   140→  readonly sigPubJwk: JsonWebKey;
   141→  readonly ecdh: SessionEcdh;
   142→  readonly registry = new Map<string, RegistryEntry>();
   143→  kv = 1;
   144→
   145→  private readonly sigPrivJwk: JsonWebKey;
   146→  private readonly entryKey: CryptoKey;
   147→  private readonly keys = new Map<number, InstalledKey>();
   148→  private readonly clock: SendClock = createSendClock();
   149→  private readonly replay: ReplayGuard;
   150→  private readonly now: () => number;
   151→  private readonly graceMs: number;
   152→  private pending: { frame: WireFrame; at: number }[] = [];
   153→  private readyQueue: WireFrame[] = [];
   154→  private readonly files = new Map<string, AssemblingFile>();
   155→  private readonly highWater: Record<string, number> = {};
   156→  private readonly watermarks?: WatermarkStore;
   157→
   158→  constructor(init: CipherInit) {
   159→    this.roomId = init.roomId;
   160→    this.selfId = init.selfId;
   161→    this.sigPrivJwk = init.sig.privJwk;
   162→    this.sigPubJwk = init.sig.pubJwk;
   163→    this.ecdh = init.ecdh;
   164→    this.entryKey = init.entryKey;
   165→    this.now = init.now ?? Date.now;
   166→    this.graceMs = init.graceMs ?? KEY_GRACE_MS;
   167→    this.watermarks = init.watermarks;
   168→    this.keys.set(1, { key: init.entryKey, raw: new Uint8Array(0), installedAt: this.now() });
   169→    if (init.watermarks) {
   170→      try {
   171→        this.highWater = init.watermarks.load() ?? {};
   172→      } catch {
   173→        this.highWater = {};
   174→      }
   175→    }
   176→    const initial = new Map<string, number>(
   177→      Object.entries(this.highWater).map(([k, v]) => [k, Number(v)]),
   178→    );
   179→    this.replay = createReplayGuard({
   180→      initialHigh: initial,
   181→      onAccept: (key, counter) => {
   182→        this.highWater[key] = counter;
   183→        this.persistWatermarks();
   184→      },
   185→    });
   186→  }
   187→
   188→  private persistWatermarks() {
   189→    if (!this.watermarks) return;
   190→    const entries = Object.entries(this.highWater);
   191→    // Bounded: keep the most recent 512 (sender, session) watermarks.
   192→    const trimmed = entries.length > 512 ? entries.slice(entries.length - 512) : entries;
   193→    try {
   194→      this.watermarks.save(Object.fromEntries(trimmed));
   195→    } catch {
   196→      /* storage full/unavailable — defense degrades to per-page-load */
   197→    }
   198→  }
   199→
   200→  /* ---------------- key ring ---------------- */
   201→
   202→  getKey(kv: number): CryptoKey | undefined {
   203→    this.pruneGrace();
   204→    return this.keys.get(kv)?.key;
   205→  }
   206→
   207→  /** Install a newer key version. Old versions stay decryptable for the
   208→   *  grace window, then are refused. Returns frames that were parked
   209→   *  pending this key (the caller should drain them). */
   210→  installKey(
   211→    kv: number,
   212→    key: CryptoKey,
   213→    opts?: { rotation?: boolean; raw?: Uint8Array },
   214→  ): WireFrame[] {
   215→    if (
   216→      kv <= this.kv ||
   217→      !Number.isSafeInteger(kv) ||
   218→      kv > KVERSION_CAP
   219→    ) {
   220→      return [];
   221→    }
   222→    this.keys.set(kv, {
   223→      key,
   224→      raw: opts?.raw ?? new Uint8Array(0),
   225→      installedAt: this.now(),
   226→    });
   227→    this.kv = kv;
   228→    const ready = this.pending.filter((p) => p.frame.kv <= kv).map((p) => p.frame);
   229→    this.pending = this.pending.filter((p) => p.frame.kv > kv);
   230→    this.readyQueue.push(...ready);
   231→    return ready;
   232→  }
   233→
   234→  /** Frames that became processable after a key install, paired with
   235→   *  their open results (the pair carries the frame id — the message id). */
   236→  async drainPending(): Promise<{ frame: WireFrame; result: OpenResult }[]> {
   237→    const out: { frame: WireFrame; result: OpenResult }[] = [];
   238→    let guard = 0;
   239→    while (this.readyQueue.length && guard++ < 128) {
   240→      const frame = this.readyQueue.shift()!;
   241→      out.push({ frame, result: await this.open(frame) });
   242→    }
   243→    return out;
   244→  }
   245→
   246→  /** The entry key (kv 1) — permanent, used to wrap join deliveries. */
   247→  getEntryKey(): CryptoKey {
   248→    return this.entryKey;
   249→  }
   250→
   251→  private pruneGrace() {
   252→    const t = this.now();
   253→    for (const [kv, inst] of this.keys) {
   254→      if (kv !== this.kv && t - inst.installedAt > this.graceMs) {
   255→        this.keys.delete(kv);
   256→      }
   257→    }
   258→  }
   259→
   260→  private requireKey(kv: number): CryptoKey {
   261→    const key = this.keys.get(kv)?.key;
   262→    if (!key) throw new Error(`no key installed for version ${kv}`);
   263→    return key;
   264→  }
   265→
   266→  /* ---------------- coordinator ---------------- */
   267→
   268→  /** Deterministic coordinator: the lowest memberId among connected
   269→   *  members (excluding the given ids — e.g. the leaver). Every client
   270→   *  computes the same answer from the same registry. */
   271→  expectedCoordinator(exclude: string[] = []): string | null {
   272→    const ids = [...this.registry.keys()].filter(
   273→      (id) => !exclude.includes(id) && this.registry.get(id)?.connected !== false,
   274→    );
   275→    if (!ids.length) return null;
   276→    ids.sort();
   277→    return ids[0];
   278→  }

   282→  private async seal(

   292→      clock: this.clock,
   293→      sigPrivJwk: this.sigPrivJwk,

   295→      frameSize,
   296→      frameId: extra?.frameId,
   297→      to: extra?.to,
   298→    });
   299→  }

   315→  /** An ink margin mark on a message. The mark glyph rides the
   316→   *  canonical-signed `text` field and the target rides `messageId` —
   317→   *  both are signature-covered, so a mark is exactly as unforgeable

   321→  }
   322→
   323→  /** Early burn of one of our own messages. Signed with the burn
   324→   *  canonical, so nobody else can retire our letters. */
   325→  async sealBurn(messageId: string): Promise<WireFrame[]> {
   326→    const burnSig = await signCanonical(
   327→      this.sigPrivJwk,
   328→      burnCanonical(this.roomId, this.selfId, messageId),
   329→    );
   330→    return [await this.seal({ kind: "burn", messageId, burnSig }, CONTROL_FRAME_BYTES)];
   331→  }
   332→
   333→  /** Proof-of-possession for a clean departure: signs the canonical
   334→   *  leave string with this room's signing key, so the server can
   335→   *  verify against the registered pubkey before writing us out —
   336→   *  memberId alone must never be enough to rotate the room. */
   337→  async signDepartureProof(ts: number = Date.now()) {
   338→    return signLeaveProof(this.sigPrivJwk, this.roomId, this.selfId, ts);
   339→  }

   341→  /** A file becomes exactly 1 control-sized meta frame plus a FIXED
   342→   *  number of uniform chunk frames — regardless of true file size. */
   343→  async sealFile(opts: {
   344→    name: string;
   345→    mime: string;
   346→    size: number;
   347→    dataB64: string;
   348→    sha: string;
   349→    /** caption typed alongside the attachment */
   350→    text?: string;
   351→    ttlSec?: number;
   352→    viewOnce?: boolean;
   353→  }): Promise<WireFrame[]> {
   354→    const padded = padB64Payload(opts.dataB64, FILE_PAYLOAD_B64);
   355→    const messageId = crypto.randomUUID();
   356→    const meta = await this.seal(
   357→      {
   358→        kind: "file:meta",
   359→        messageId,
   360→        // Top-level text is part of canonicalV2 — the caption is signed,
   361→        // so it is exactly as authentic as the sender's words in text frames.
   362→        text: opts.text,
   363→        file: {
   364→          messageId,
   365→          name: opts.name,
   366→          mime: opts.mime,
   367→          size: opts.size,
   368→          sha: opts.sha,
   369→          ttlSec: opts.ttlSec,
   370→          viewOnce: opts.viewOnce,
   371→        },
   372→      },
   373→      CONTROL_FRAME_BYTES,
   374→      { frameId: messageId },
   375→    );
   376→    const frames: WireFrame[] = [meta];
   377→    for (let seq = 0; seq < FILE_TOTAL_CHUNKS; seq++) {
   378→      const dataB64 = padded.slice(seq * FILE_CHUNK_B64, (seq + 1) * FILE_CHUNK_B64);
   379→      frames.push(
   380→        await this.seal(
   381→          { kind: "file:chunk", chunk: { messageId, seq, total: FILE_TOTAL_CHUNKS, dataB64 } },
   382→          FILE_FRAME_BYTES,
   383→        ),
   384→      );
   385→    }
   386→    return frames;
   387→  }

   389→  /** Wrap a room key for one recipient over ECDH. Rotation offers ride
   390→   *  in a frame sealed under the OLD key (recipients can still open
   391→   *  it); join deliveries ride in a frame sealed under the ENTRY key
   392→   *  (only a joiner who proved the password can open it) and carry a
   393→   *  second, entry-key layer around the ECDH wrap. */
   394→  async sealKeyOffer(opts: {
   395→    to: string;
   396→    kv: number;
   397→    keyRaw: Uint8Array;
   398→    rot: boolean;
   399→    outerKey?: CryptoKey;
   400→    wireKey?: CryptoKey;
   401→    wireKv?: number;
   402→    overrideEcdh?: SessionEcdh;
   403→  }): Promise<WireFrame[]> {
   404→    const ecdhUse = opts.overrideEcdh ?? this.ecdh;
   405→    const theirPub = this.registry.get(opts.to)?.ecdhPubB64;
   406→    if (!theirPub) return []; // nothing we can do for a member without a session key
   407→    const { wrapIv, wrapCt } = await ecdhWrapKey({
   408→      myPrivJwk: ecdhUse.privJwk,
   409→      theirPubRawB64: theirPub,
   410→      roomId: this.roomId,
   411→      kv: opts.kv,
   412→      payload: { keyB64: toB64(opts.keyRaw), kv: opts.kv, rot: opts.rot, issuedAt: this.now() },
   413→    });
   414→    let offer: OfferBody = {
   415→      to: opts.to,
   416→      kv: opts.kv,
   417→      rot: opts.rot,
   418→      ecdhPubB64: ecdhUse.pubRawB64,
   419→      wrapIv,
   420→      wrapCt,
   421→    };
   422→    if (opts.outerKey) {
   423→      const outer = await encryptJson(opts.outerKey, {
   424→        ecdhPubB64: ecdhUse.pubRawB64,
   425→        wrapIv,
   426→        wrapCt,
   427→      });
   428→      offer = { to: opts.to, kv: opts.kv, rot: opts.rot, outerIv: outer.iv, outerCt: outer.ct };
   429→    }
   430→    return [
   431→      await this.seal(
   432→        { kind: "key:offer", messageId: `offer:${opts.to}:${opts.kv}`, offer },
   433→        CONTROL_FRAME_BYTES,
   434→        {
   435→          to: opts.to,
   436→          wireKey: opts.wireKey ?? (opts.outerKey ? this.entryKey : undefined),
   437→          wireKv: opts.wireKv ?? (opts.outerKey || opts.rot ? 0 : undefined),
   438→        },
   439→      ),
   440→    ];
   441→  }
   442→
   443→  /** Rotate unconditionally: generate a RANDOM new room key and
   444→   *  deliver it pairwise over ECDH. Callers decide whether they have
   445→   *  the right to rotate (see rotateAsCoordinator / the rejoin
   446→   *  fallback in the store — simultaneous fallbacks converge because
   447→   *  version offers are monotonic).
   448→   *
   449→   *  `rot` marks the offer as the formal leave-ceremony (receivers
   450→   *  verify the sender is their coordinator). Fallback re-seals use
   451→   *  rot=false — a member-initiated delivery, still ECDH-wrapped,
   452→   *  signature-checked, registry-gated and version-monotonic. */
   453→  async rotateTo(minKv?: number, rot = true): Promise<WireFrame[]> {
   454→    const newKv = Math.max(this.kv, minKv ?? 0) + 1;
   455→    const keyRaw = crypto.getRandomValues(new Uint8Array(32));
   456→    const key = await importAesKey(keyRaw);
   457→    this.keys.set(newKv, { key, raw: keyRaw, installedAt: this.now() });
   458→    this.kv = newKv;
   459→
   460→    const frames: WireFrame[] = [];
   461→    for (const [id, entry] of this.registry) {
   462→      if (id === this.selfId || !entry.ecdhPubB64) continue;
   463→      frames.push(
   464→        ...(await this.sealKeyOffer({
   465→          to: id,
   466→          kv: newKv,
   467→          keyRaw,
   468→          rot,
   469→          wireKey: this.entryKey,
   470→          wireKv: 0,
   471→        })),
   472→      );
   473→    }
   474→    return frames;
   475→  }
   476→
   477→  /** Rotate IF this client is the deterministic coordinator (the
   478→   *  normal ceremony after a member leaves).
   479→   *
   480→   *  `minKv` is the server's rotation ledger (room epoch): a member
   481→   *  re-joining after a refresh may have lost count of rotations, and a
   482→   *  departed member may hold keys up to that version — so the new
   483→   *  version must exceed it. Offers are sealed under the ENTRY key so
   484→   *  every member can read them no matter which version they hold. */
   485→  async rotateAsCoordinator(minKv?: number): Promise<WireFrame[]> {
   486→    if (this.expectedCoordinator() !== this.selfId) return [];
   487→    return this.rotateTo(minKv);
   488→  }
   489→
   490→  /** Deliver the CURRENT room key to a new member — ECDH-wrapped and
   491→   *  additionally sealed under the entry key, so only a joiner who
   492→   *  derived the same password can open it. No-op while the room is
   493→   *  still on its password-derived key (joiners derive that themselves). */
   494→  async deliverKeyTo(memberId: string): Promise<WireFrame[]> {
   495→    if (this.kv <= 1) return [];
   496→    const inst = this.keys.get(this.kv);
   497→    if (!inst || !inst.raw.length) return [];
   498→    return this.sealKeyOffer({
   499→      to: memberId,
   500→      kv: this.kv,
   501→      keyRaw: inst.raw,
   502→      rot: false,
   503→      outerKey: this.entryKey,
   504→    });
   505→  }
   506→
   507→  /* ---------------- opening ---------------- */
   508→
   509→  async open(frame: WireFrame): Promise<OpenResult> {
   510→    if (
   511→      !frame ||
   512→      frame.v !== 2 ||
   513→      typeof frame.ct !== "string" ||
   514→      typeof frame.iv !== "string" ||
   515→      typeof frame.from !== "string"
   516→    ) {
   517→      return { type: "reject", reason: "shape" };
   518→    }
   519→    // Addressed to someone else (the relay broadcasts everything).
   520→    if (frame.to && frame.to !== this.selfId) {
   521→      return { type: "reject", reason: "offer" };
   522→    }
   523→    // Eviction gate: frames only flow between CURRENT members.
   524→    const entry = this.registry.get(frame.from);
   525→    if (!entry) {
   526→      return { type: "reject", reason: "registry" };
   527→    }
   528→
   529→    // Key selection.
   530→    this.pruneGrace();
   531→    let key: CryptoKey | undefined;
   532→    if (frame.kv === 0) {
   533→      // Entry-sealed frame (key-delivery offer) — the entry key is
   534→      // permanent, so these stay readable at any key version. The
   535→      // secret inside is still wrapped pairwise over ECDH.
   536→      key = this.entryKey;
   537→    } else if (frame.kv === this.kv) {
   538→      key = this.keys.get(frame.kv)?.key;
   539→      if (!key) return { type: "reject", reason: "kv" };
   540→    } else if (frame.kv > this.kv) {
   541→      // Newer key version: either a rotation is in flight (we are one
   542→      // offer behind) or we are a joiner awaiting delivery. Hold fresh
   543→      // frames briefly; expire squatters; refuse anything stale.
   544→      const t = this.now();
   545→      this.pending = this.pending.filter((p) => t - p.at <= PENDING_TTL_MS);
   546→      if (t - (frame.ts ?? 0) <= PENDING_TTL_MS && this.pending.length < PENDING_MAX) {
   547→        this.pending.push({ frame, at: t });
   548→        return { type: "pending" };
   549→      }
   550→      return { type: "reject", reason: "kv" };
   551→    } else {
   552→      key = this.keys.get(frame.kv)?.key; // grace window for old versions
   553→      if (!key) return { type: "reject", reason: "kv" };
   554→    }
   555→
   556→    const body = await openFrame(key, frame);
   557→    if (!body) return { type: "reject", reason: "decrypt" };
   558→
   559→    // Signature against the sender's REGISTERED key.
   560→    const canonical = canonicalV2({
   561→      roomId: this.roomId,
   562→      kv: frame.kv,
   563→      senderId: frame.from,
   564→      sessionTag: body.sessionTag,
   565→      counter: body.counter,
   566→      ts: body.ts,
   567→      kind: body.kind,
   568→      text: body.text,
   569→      fileSha: body.fileSha,
   570→      messageId: body.messageId,
   571→    });
   572→    const okSig = await verifyCanonical(entry.pubkey, canonical, body.sig);
   573→    if (!okSig) return { type: "reject", reason: "signature" };
   574→
   575→    // Replay: counters, timestamp window, frame-id dedup.
   576→    const okReplay = this.replay.check({
   577→      senderId: frame.from,
   578→      sessionTag: body.sessionTag,
   579→      counter: body.counter,
   580→      ts: body.ts,
   581→      id: frame.id,
   582→    });
   583→    if (!okReplay) return { type: "reject", reason: "replay" };
   584→
   585→    switch (body.kind) {
   586→      case "text":
   587→        return { type: "text", body };
   588→      case "typing":
   589→        return { type: "typing", senderId: frame.from };
   590→      case "spent":
   591→        if (!body.messageId) return { type: "reject", reason: "shape" };
   592→        return { type: "spent", messageId: body.messageId, senderId: frame.from };
   593→      case "burn": {
   594→        if (!body.messageId || !body.burnSig) return { type: "reject", reason: "shape" };
   595→        const okBurn = await verifyCanonical(
   596→          entry.pubkey,
   597→          burnCanonical(this.roomId, frame.from, body.messageId),
   598→          body.burnSig,
   599→        );
   600→        if (!okBurn) return { type: "reject", reason: "signature" };
   601→        return { type: "burn", messageId: body.messageId, senderId: frame.from };
   602→      }
   603→      case "react": {
   604→        // A mark is only valid if it is one of the product's four glyphs —
   605→        // the wire never carries arbitrary strings a page could abuse.
   606→        if (!body.messageId || typeof body.text !== "string" || !isReactionMark(body.text)) {
   607→          return { type: "reject", reason: "shape" };
   608→        }
   609→        return { type: "react", messageId: body.messageId, senderId: frame.from, mark: body.text };
   610→      }
   611→      case "file:meta": {
   612→        if (!body.file?.messageId) return { type: "reject", reason: "shape" };
   613→        // Bounded: a flood of orphan metas cannot grow the map forever.
   614→        if (this.files.size > 32) {
   615→          const oldest = this.files.keys().next().value;
   616→          if (oldest !== undefined) this.files.delete(oldest);
   617→        }
   618→        this.files.set(body.file.messageId, {
   619→          meta: body.file,
   620→          text: typeof body.text === "string" && body.text.length > 0 ? body.text : undefined,
   621→          metaTs: body.ts,
   622→          chunks: new Map(),
   623→        });
   624→        return { type: "file-meta", senderId: frame.from };
   625→      }
   626→      case "file:chunk": {
   627→        const c = body.chunk;
   628→        if (!c?.messageId || typeof c.seq !== "number" || typeof c.total !== "number") {
   629→          return { type: "reject", reason: "shape" };
   630→        }
   631→        const file = this.files.get(c.messageId);
   632→        if (!file) return { type: "reject", reason: "shape" };
   633→        file.chunks.set(c.seq, c.dataB64 ?? "");
   634→        if (file.chunks.size < c.total) return { type: "file-chunk", senderId: frame.from };
   635→        // Complete — assemble, verify, deliver.
   636→        this.files.delete(c.messageId);
   637→        let padded = "";
   638→        for (let i = 0; i < c.total; i++) padded += file.chunks.get(i) ?? "";
   639→        const realLength = Math.ceil(file.meta.size / 3) * 4;
   640→        const dataB64 = padded.slice(0, realLength);
   641→        try {
   642→          const sha = await sha256HexOfBytes(fromB64(dataB64));
   643→          if (sha !== file.meta.sha) return { type: "reject", reason: "sha" };
   644→        } catch {
   645→          return { type: "reject", reason: "sha" };
   646→        }
   647→        return {
   648→          type: "file",
   649→          senderId: frame.from,
   650→          ts: file.metaTs,
   651→          text: file.text,
   652→          file: { ...file.meta, dataB64 },
   653→        };
   654→      }
   655→      case "key:offer":
   656→        return this.openOffer(body.offer, frame.from);
   657→      default:
   658→        return { type: "reject", reason: "shape" };
   659→    }
   660→  }
   661→
   662→  private async openOffer(offer: OfferBody | undefined, senderId: string): Promise<OpenResult> {
   663→    if (!offer || offer.to !== this.selfId || typeof offer.kv !== "number") {
   664→      return { type: "reject", reason: "offer" };
   665→    }
   666→    // Unwrap the entry-key layer (join deliveries only).
   667→    let inner: { ecdhPubB64?: string; wrapIv?: string; wrapCt?: string } = offer;
   668→    if (offer.outerIv && offer.outerCt) {
   669→      const opened = await decryptJson<{ ecdhPubB64?: string; wrapIv?: string; wrapCt?: string }>(
   670→        this.entryKey,
   671→        offer.outerIv,
   672→        offer.outerCt,
   673→      );
   674→      if (!opened?.ecdhPubB64 || !opened.wrapIv || !opened.wrapCt) {
   675→        return { type: "reject", reason: "offer" };
   676→      }
   677→      inner = opened;
   678→    }
   679→    if (!inner.ecdhPubB64 || !inner.wrapIv || !inner.wrapCt) {
   680→      return { type: "reject", reason: "offer" };
   681→    }
   682→    // Unknown-key-share defense: the ECDH key must be the one this
   683→    // sender has in our registry.
   684→    const regEcdh = this.registry.get(senderId)?.ecdhPubB64;
   685→    if (!regEcdh || inner.ecdhPubB64 !== regEcdh) {
   686→      return { type: "reject", reason: "offer" };
   687→    }
   688→    // Rotation offers may only come from the coordinator.
   689→    if (offer.rot && senderId !== this.expectedCoordinator()) {
   690→      return { type: "reject", reason: "offer" };
   691→    }
   692→    const payload = await ecdhUnwrapKey({
   693→      myPrivJwk: this.ecdh.privJwk,
   694→      theirPubRawB64: inner.ecdhPubB64,
   695→      roomId: this.roomId,
   696→      kv: offer.kv,
   697→      wrapIv: inner.wrapIv,
   698→      wrapCt: inner.wrapCt,
   699→    });
   700→    if (
   701→      !payload ||
   702→      typeof payload.keyB64 !== "string" ||
   703→      payload.kv !== offer.kv ||
   704→      typeof offer.kv !== "number" ||
   705→      !Number.isSafeInteger(offer.kv) ||
   706→      offer.kv > KVERSION_CAP ||
   707→      offer.kv <= this.kv
   708→    ) {
   709→      return { type: "reject", reason: "offer" };
   710→    }
   711→    const keyRaw = fromB64(payload.keyB64);
   712→    const key = await importAesKey(keyRaw);
   713→    this.installKey(offer.kv, key, { rotation: !!offer.rot, raw: keyRaw });
   714→    return { type: "offer-installed", kv: offer.kv, rotation: !!offer.rot };
   715→  }
   716→}
