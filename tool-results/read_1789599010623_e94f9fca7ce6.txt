   138→   138→function pruneTypingLater(roomId: string) {
   139→   139→  const existing = typingPruneTimers.get(roomId);
   140→   140→  if (existing) clearTimeout(existing);
   141→   141→  const t = setTimeout(() => {
   142→   142→    typingPruneTimers.delete(roomId);
   143→   143→    useApp.setState((s) => {
   144→   144→      const fresh = (s.typing[roomId] ?? []).filter(
   145→   145→        (x) => Date.now() - x.at < TYPING_TTL_MS,
   146→   146→      );
   147→   147→      return { typing: { ...s.typing, [roomId]: fresh } };
   148→   148→    });
   149→   149→  }, TYPING_TTL_MS + 100);
   150→   150→  typingPruneTimers.set(roomId, t);
   151→   151→}
   152→   152→
   153→   153→function clearTyping(roomId: string, memberId: string) {
   154→   154→  useApp.setState((s) => {
   155→   155→    const list = s.typing[roomId] ?? [];
   156→   156→    if (!list.some((t) => t.memberId === memberId)) return {};
   157→   157→    return {
   158→   158→      typing: {
   159→   159→        ...s.typing,
   160→   160→        [roomId]: list.filter((t) => t.memberId !== memberId),
   161→   161→      },
   162→   162→    };
   163→   163→  });
   164→   164→}
   165→   165→
   166→   166→function scheduleBurn(
   167→   167→  roomId: string,
   168→   168→  messageId: string,
   169→   169→  expiresAt: number,
   170→   170→  onBurn: (roomId: string, messageId: string) => void,
   171→   171→) {
   172→   172→  if (burnTimers.has(messageId)) return;
   173→   173→  const delay = Math.max(0, expiresAt - Date.now());
   174→   174→  const t = setTimeout(() => {
   175→   175→    burnTimers.delete(messageId);
   176→   176→    onBurn(roomId, messageId);
   177→   177→  }, delay);
   178→   178→  burnTimers.set(messageId, t);
   179→   179→}
   180→   180→
   181→   181→function cancelBurns(roomId: string, messages: MessageView[]) {
   182→   182→  for (const m of messages) {
   183→   183→    if (m.id) {
   184→   184→      const t = burnTimers.get(m.id);
   185→   185→      if (t) {
   186→   186→        clearTimeout(t);
   187→   187→        burnTimers.delete(m.id);
   188→   188→      }
   189→   189→    }
   190→   190→  }
   191→   191→  void roomId;
   192→   192→}

   214→   214→  init: async () => {
   215→   215→    if (get().ready) return;
   216→   216→    // One random seed per device; every room derives its own signing
   217→   217→    // key from it, so registries can never be correlated across rooms.
   218→   218→    const seed = await loadDeviceSeed();
   219→   219→    // Burned rooms showed their ash last session; now they are gone.
   220→   220→    const cards = sweepBurnedRooms();
   221→   221→    set({ seed, roomCards: cards, ready: true });
   222→   222→
   223→   223→    get().syncHash();

   225→   225→

   228→   228→
   229→   229→  syncHash: () => {
   230→   230→    const hash = window.location.hash.replace(/^#\/?/, "");
   231→   231→    if (hash.startsWith("join")) {
   232→   232→      const code = hash.split("/")[1] ?? "";
   233→   233→      set({ screen: "invite", inviteCode: code || null, activeRoomId: null });
   234→   234→      return;
   235→   235→    }
   236→   236→    if (hash.startsWith("new")) {
   237→   237→      set({ screen: "landing", activeRoomId: null });
   238→   238→      return;
   239→   239→    }
   240→   240→    if (hash.startsWith("rooms")) {
   241→   241→      set({ screen: "rooms", activeRoomId: null });
   242→   242→      return;
   243→   243→    }
   244→   244→    if (hash.startsWith("r/")) {
   245→   245→      const roomId = hash.slice(2);
   246→   246→      const session = getSession(roomId);
   247→   247→      set({ screen: "chat", activeRoomId: roomId });
   248→   248→      if (session) {
   249→   249→        patchRoomCard(roomId, { unread: false });
   250→   250→        set({ roomCards: loadRoomCards() });
   251→   251→      }
   252→   252→      // No keys in memory — the room is locked. A first-class state,
   253→   253→      // never an error.
   254→   254→      return;
   255→   255→    }
   256→   256→    // Root: the desk if you have rooms, the landing if you don't.
   257→   257→    if (get().roomCards.length > 0) {
   258→   258→      set({ screen: "rooms", activeRoomId: null });
   259→   259→    } else {
   260→   260→      set({ screen: "landing", activeRoomId: null });
   261→   261→    }
   262→   262→  },
   263→   263→
   264→   264→  navigate: (screen, roomId = null) => {

   266→   266→      screen === "rooms"
   267→   267→        ? "#/rooms"
   268→   268→        : screen === "chat" && roomId
   269→   269→          ? `#/r/${roomId}`
   270→   270→          : screen === "invite"
   271→   271→            ? "#/join"
   272→   272→            : screen === "landing"
   273→   273→              ? "#/new"
   274→   274→              : "#/";

   276→   276→      window.location.hash = hash;
   277→   277→    } else {
   278→   278→      get().syncHash();
   279→   279→    }

   356→   356→  /* ------------------------------------------------- join ---- */
   357→   357→
   358→   358→  joinRoom: async (code, password, localName) => {
   359→   359→    const seed = get().seed;
   360→   360→    if (!seed) return { ok: false, reason: "error" };
   361→   361→
   362→   362→    const infoRes = await fetch(`/api/rooms/${encodeURIComponent(code)}`);
   363→   363→    if (!infoRes.ok) {
   364→   364→      const info = await infoRes.json().catch(() => ({}));
   365→   365→      return {
   366→   366→        ok: false,
   367→   367→        reason: info?.burned ? "burned" : "not-found",
   368→   368→      };
   369→   369→    }
   370→   370→    const info = await infoRes.json();
   371→   371→    if (info.memberCount >= info.memberCap) {
   372→   372→      return { ok: false, reason: "room-full" };
   373→   373→    }
   374→   374→
   375→   375→    set({ sealing: { roomId: code, label: "Sealing the room" } });
   376→   376→    const started = Date.now();
   377→   377→    // The entry key (version 1) — argon2id for new rooms, PBKDF2 for
   378→   378→    // rooms created before the upgrade. The bundle decides.
   379→   379→    const entryKey = await unlockWithBundle(password, info.verifier, code, info.epoch);
   380→   380→    await new Promise((r) =>
   381→   381→      setTimeout(r, Math.max(0, MIN_SEAL_MS - (Date.now() - started))),
   382→   382→    );
   383→   383→
   384→   384→    if (!entryKey) {
   385→   385→      set({ sealing: null });
   386→   386→      return { ok: false, reason: "wrong-password" };
   387→   387→    }
   388→   388→
   389→   389→    const [identity, ecdh] = await Promise.all([
   390→   390→      deriveRoomSigningKey(seed, code),
   391→   391→      generateSessionEcdh(),
   392→   392→    ]);
   393→   393→    const alias = aliasFromFingerprint(identity.fingerprintHex);
   394→   394→    const colorIdx = inkFromFingerprint(identity.fingerprintHex);
   395→   395→    const joinRes = await fetch(`/api/rooms/${code}/members`, {
   396→   396→      method: "POST",
   397→   397→      headers: { "content-type": "application/json" },
   398→   398→      body: JSON.stringify({
   399→   399→        pubkey: identity.pubJwk,
   400→   400→        ecdhPub: ecdh.pubRawB64,
   401→   401→        alias,
   402→   402→        colorIdx,
   403→   403→      }),
   404→   404→    });
   405→   405→    if (!joinRes.ok) {
   406→   406→      set({ sealing: null });
   407→   407→      if (joinRes.status === 403) return { ok: false, reason: "room-full" };
   408→   408→      return { ok: false, reason: "error" };
   409→   409→    }
   410→   410→    const { memberId, rejoined } = await joinRes.json();
   411→   411→
   412→   412→    const cipher = new RoomCipher({
   413→   413→      roomId: code,
   414→   414→      selfId: memberId,
   415→   415→      sig: { privJwk: identity.privJwk, pubJwk: identity.pubJwk },
   416→   416→      ecdh,
   417→   417→      entryKey,
   418→   418→    });
   419→   419→    setCipher(code, cipher);
   420→   420→
   421→   421→    const session: RoomSession = {
   422→   422→      roomId: code,
   423→   423→      memberId,
   424→   424→      alias,
   425→   425→      colorIdx,
   426→   426→      kv: 1, // the current key arrives via ECDH offers if the room has rotated
   427→   427→      password,
   428→   428→      creatorToken: loadCreatorToken(code),
   429→   429→      defaultTtl: loadRoomSettings(code).defaultTtl,
   430→   430→    };
   431→   431→    setSession(session);
   432→   432→    set({ sealing: null });
   433→   433→    const name =
   434→   434→      localName ||
   435→   435→      get().roomCards.find((c) => c.roomId === code)?.localName ||
   436→   436→      `Room ${code.slice(0, 4)}`;
   437→   437→    await get().enterRoom(session, name, !!rejoined);
   438→   438→
   439→   439→    // The server's epoch is the rotation LEDGER (it persists; keys do
   440→   440→    // not). If the room has rotated before and we are back at the
   441→   441→    // password-derived key — a rejoin after refresh, or a fresh join —
   442→   442→    // the current key must arrive via ECDH offers. Give live members a
   443→   443→    // moment to deliver; if nobody does (everyone refreshed), the
   444→   444→    // coordinator re-seals past the ledger so keys any departed member
   445→   445→    // may still hold are dead again.
   446→   446→    if (info.epoch > 1) {
   447→   447→      const attemptRotation = (minKv: number) => {
   448→   448→        const c = getCipher(code);
   449→   449→        const s = getSession(code);
   450→   450→        if (!c || !s || c.kv > 1) return;
   451→   451→        void c.rotateTo(minKv).then((offers) => {
   452→   452→          const relay2 = getRelay();
   453→   453→          for (const f of offers) {
   454→   454→            relay2.emit("message:send", { roomId: code, envelope: f });
   455→   455→          }
   456→   456→          addSystemLine(code, "The room re-sealed for those who remain.");
   457→   457→        });
   458→   458→      };
   459→   459→      setTimeout(() => {
   460→   460→        const cipher2 = getCipher(code);
   461→   461→        const session2 = getSession(code);
   462→   462→        if (!cipher2 || !session2 || cipher2.kv > 1) return;
   463→   463→        if (cipher2.expectedCoordinator() === session2.memberId) {
   464→   464→          attemptRotation(info.epoch);
   465→   465→        } else {
   466→   466→          // Ask the room for the current key; a live key-holder answers
   467→   467→          // with an ECDH offer.
   468→   468→          getRelay().emit("key:request", { roomId: code });
   469→   469→          setResealing(code, true);
   470→   470→          // Fallback: if nobody answers (every member refreshed, or the
   471→   471→          // holder is offline), re-seal on our own after a randomized
   472→   472→          // delay. The version jump is randomized too, so simultaneous
   473→   473→          // fallbacks converge: a lower-version holder always accepts
   474→   474→          // a higher-version offer.
   475→   475→          setTimeout(() => {
   476→   476→            const c = getCipher(code);
   477→   477→            const s = getSession(code);
   478→   478→            if (!c || !s || c.kv > 1) return;
   479→   479→            void c.rotateTo(info.epoch + 1 + Math.floor(Math.random() * 8), false).then((offers) => {
   480→   480→              const relay2 = getRelay();
   481→   481→              for (const f of offers) {
   482→   482→                relay2.emit("message:send", { roomId: code, envelope: f });
   483→   483→              }
   484→   484→              addSystemLine(code, "The room re-sealed for those who remain.");
   485→   485→              setResealing(code, false);
   486→   486→            });
   487→   487→          }, 5000 + Math.floor(Math.random() * 3000));
   488→   488→        }
   489→   489→      }, 3000);
   490→   490→    }
   491→   491→    return { ok: true };
   492→   492→  },

   531→   531→  refreshMembers: async (roomId) => {
   532→   532→    try {
   533→   533→      const res = await fetch(`/api/rooms/${roomId}/members`);
   534→   534→      if (!res.ok) return;
   535→   535→      const { members } = (await res.json()) as { members: MemberPublic[] };
   536→   536→      set((s) => ({ members: { ...s.members, [roomId]: members } }));
   537→   537→
   538→   538→      // Keep the cipher's registry in sync — it is the eviction gate.
   539→   539→      const cipher = getCipher(roomId);
   540→   540→      if (cipher) {
   541→   541→        const seen = new Set<string>();
   542→   542→        for (const m of members) {
   543→   543→          seen.add(m.memberId);
   544→   544→          const cur = cipher.registry.get(m.memberId);
   545→   545→          cipher.registry.set(m.memberId, {
   546→   546→            pubkey: m.pubkey,
   547→   547→            ecdhPubB64: m.ecdhPub ?? cur?.ecdhPubB64,
   548→   548→            connected: cur?.connected ?? true,
   549→   549→            joinedAt: m.joinedAt ?? cur?.joinedAt,
   550→   550→            alias: m.alias,
   551→   551→            colorIdx: m.colorIdx,
   552→   552→          });
   553→   553→        }
   554→   554→        for (const id of [...cipher.registry.keys()]) {
   555→   555→          if (!seen.has(id)) cipher.registry.delete(id);
   556→   556→        }
   557→   557→      }
   558→   558→    } catch {
   559→   559→      /* offline — the registry stays as-is */
   560→   560→    }
   561→   561→  },

   565→   565→  sendMessage: async (text, file, ttlOverride) => {
   566→   566→    const viewOnce = !!file?.viewOnce;
   567→   567→    const { activeRoomId } = get();
   568→   568→    if (!activeRoomId) return;
   569→   569→    const session = getSession(activeRoomId);
   570→   570→    const cipher = getCipher(activeRoomId);
   571→   571→    if (!session || !cipher) return;
   572→   572→    const trimmed = text.trim();
   573→   573→    if (!trimmed && !file) return;
   574→   574→
   575→   575→    const ttlSec = (ttlOverride !== undefined ? ttlOverride : session.defaultTtl) || undefined;
   576→   576→    const ts = Date.now();
   577→   577→
   578→   578→    let frames: WireFrame[];
   579→   579→    if (file) {
   580→   580→      const sha = await sha256HexOfBytes(fromB64(file.dataB64));
   581→   581→      frames = await cipher.sealFile({
   582→   582→        name: file.name,
   583→   583→        mime: file.mime,
   584→   584→        size: file.size,
   585→   585→        dataB64: file.dataB64,
   586→   586→        sha,
   587→   587→        ttlSec,
   588→   588→        viewOnce: viewOnce || undefined,
   589→   589→      });
   590→   590→    } else {
   591→   591→      frames = await cipher.sealText({ text: trimmed, ttlSec });
   592→   592→    }
   593→   593→    // The first frame's id IS the message id (text frame, or the file
   594→   594→    // meta frame) — the relay acks it and the view flips to "sent".
   595→   595→    const id = frames[0].id;
   596→   596→
   597→   597→    const kind = file ? "file" : "text";
   598→   598→    const view: MessageView = {
   599→   599→      id,
   600→   600→      kind,
   601→   601→      senderId: session.memberId,
   602→   602→      senderAlias: session.alias,
   603→   603→      senderColor: session.colorIdx,
   604→   604→      self: true,
   605→   605→      status: "sending",
   606→   606→      ts,
   607→   607→      text: trimmed,
   608→   608→      file: file
   609→   609→        ? { name: file.name, mime: file.mime, size: file.size, dataB64: file.dataB64 }
   610→   610→        : undefined,
   611→   611→      ttlSec,
   612→   612→      expiresAt: ttlSec ? ts + ttlSec * 1000 : undefined,
   613→   613→      viewOnce: file && viewOnce ? true : undefined,
   614→   614→    };
   615→   615→
   616→   616→    set((s) => ({
   617→   617→      messages: {
   618→   618→        ...s.messages,
   619→   619→        [session.roomId]: [...(s.messages[session.roomId] ?? []), view],
   620→   620→      },
   621→   621→    }));
   622→   622→    patchRoomCard(session.roomId, { lastActivity: ts });
   623→   623→    set({ roomCards: loadRoomCards() });
   624→   624→
   625→   625→    if (ttlSec) {
   626→   626→      scheduleBurn(session.roomId, id, ts + ttlSec * 1000, onMessageBurn);
   627→   627→    }
   628→   628→
   629→   629→    const relay = getRelay();
   630→   630→    for (const f of frames) {
   631→   631→      relay.emit("message:send", { roomId: session.roomId, envelope: f });
   632→   632→    }
   633→   633→  },
   634→   634→
   635→   635→  /* --------------------------------------------- typing ------- */
   636→   636→
   637→   637→  emitTyping: (roomId) => {
   638→   638→    const session = getSession(roomId);
   639→   639→    const cipher = getCipher(roomId);
   640→   640→    if (!session || !cipher) return;
   641→   641→    const now = Date.now();
   642→   642→    const last = typingLastEmit.get(roomId) ?? 0;

   645→   645→    // Typing rides the same uniform encrypted frames as everything
   646→   646→    // else — the relay cannot even tell WHEN someone is typing.
   647→   647→    void cipher.sealTyping().then((frames) => {
   648→   648→      const relay = getRelay();
   649→   649→      for (const f of frames) relay.emit("message:send", { roomId, envelope: f });
   650→   650→    });
   651→   651→  },

   679→   679→  burnMessage: async (roomId, messageId) => {

   688→   688→    const relay = getRelay();

   856→   856→function wireRelay(
   857→   857→  set: (
   858→   858→    partial:
   859→   859→      | Partial<AppState>
   860→   860→      | ((state: AppState) => Partial<AppState>),
   861→   861→  ) => void,
   862→   862→  get: () => AppState,
   863→   863→) {
   864→   864→  const relay = getRelay();
   865→   865→
   866→   866→  relay.on("connect", () => {
   867→   867→    set({ relayOnline: true });
   868→   868→    for (const roomId of sessionRoomIds()) {
   869→   869→      const session = getSession(roomId);
   870→   870→      const cipher = getCipher(roomId);
   871→   871→      if (session && cipher) {
   872→   872→        relay.emit("room:join", {
   873→   873→          roomId,
   874→   874→          memberId: session.memberId,
   875→   875→          alias: session.alias,
   876→   876→          colorIdx: session.colorIdx,
   877→   877→          pubkey: cipher.sigPubJwk,
   878→   878→          ecdhPub: cipher.ecdh.pubRawB64,
   879→   879→          rejoined: true,
   880→   880→        });
   881→   881→      }
   882→   882→    }
   883→   883→  });
   884→   884→
   885→   885→  relay.on("disconnect", () => set({ relayOnline: false }));
   886→   886→
   887→   887→  relay.on("room:state", ({ roomId, members }: { roomId: string; members: MemberPublic[] }) => {
   888→   888→    // merge live presence into the registry
   889→   889→    const registry = useApp.getState().members[roomId] ?? [];
   890→   890→    const byId = new Map(registry.map((m) => [m.memberId, { ...m }]));
   891→   891→    for (const live of members) {
   892→   892→      const existing = byId.get(live.memberId);
   893→   893→      byId.set(live.memberId, {
   894→   894→        ...live,
   895→   895→        ...(existing?.pubkey ? { pubkey: existing.pubkey } : {}),
   896→   896→      });
   897→   897→    }
   898→   898→    set((s) => ({ members: { ...s.members, [roomId]: Array.from(byId.values()) } }));
   899→   899→
   900→   900→    const cipher = getCipher(roomId);
   901→   901→    if (cipher) {
   902→   902→      for (const live of members) {
   903→   903→        const cur = cipher.registry.get(live.memberId);
   904→   904→        if (cur || live.pubkey) {
   905→   905→          cipher.registry.set(live.memberId, {
   906→   906→            pubkey: live.pubkey ?? cur?.pubkey ?? ({} as JsonWebKey),
   907→   907→            ecdhPubB64: live.ecdhPub ?? cur?.ecdhPubB64,
   908→   908→            connected: true,
   909→   909→            joinedAt: cur?.joinedAt,
   910→   910→            alias: live.alias,
   911→   911→            colorIdx: live.colorIdx,
   912→   912→          });
   913→   913→        }
   914→   914→      }
   915→   915→    }
   916→   916→  });
   917→   917→
   918→   918→  relay.on(
   919→   919→    "member:joined",
   920→   920→    ({ roomId, member, rejoined }: { roomId: string; member: MemberPublic; rejoined?: boolean }) => {
   921→   921→      const session = getSession(roomId);
   922→   922→      const cipher = getCipher(roomId);
   923→   923→      if (!session) return;
   924→   924→      set((s) => {
   925→   925→        const registry = s.members[roomId] ?? [];
   926→   926→        if (registry.some((m) => m.memberId === member.memberId)) {
   927→   927→          return {
   928→   928→            members: {
   929→   929→              ...s.members,
   930→   930→              [roomId]: registry.map((m) =>
   931→   931→                m.memberId === member.memberId ? { ...m, connected: true } : m,
   932→   932→              ),
   933→   933→            },
   934→   934→          };
   935→   935→        }
   936→   936→        return {
   937→   937→          members: { ...s.members, [roomId]: [...registry, { ...member, connected: true }] },
   938→   938→        };
   939→   939→      });
   940→   940→      if (cipher) {
   941→   941→        const cur = cipher.registry.get(member.memberId);
   942→   942→        cipher.registry.set(member.memberId, {
   943→   943→          pubkey: member.pubkey ?? cur?.pubkey ?? ({} as JsonWebKey),
   944→   944→          ecdhPubB64: member.ecdhPub ?? cur?.ecdhPubB64,
   945→   945→          connected: true,
   946→   946→          joinedAt: cur?.joinedAt,
   947→   947→          alias: member.alias,
   948→   948→          colorIdx: member.colorIdx,
   949→   949→        });
   950→   950→        // If the room has rotated past the password-derived key, hand
   951→   951→        // the newcomer the CURRENT key — ECDH-wrapped and sealed under
   952→   952→        // the entry key, so only a joiner who proved the password can
   953→   953→        // open it. Every member sends; the joiner keeps the highest.
   954→   954→        if (cipher.kv > 1 && member.memberId !== session.memberId) {
   955→   955→          void cipher.deliverKeyTo(member.memberId).then((frames) => {
   956→   956→            for (const f of frames) {
   957→   957→              relay.emit("message:send", { roomId, envelope: f });

  1134→  1134→async function receiveFrame(roomId: string, frame: WireFrame) {
  1135→  1135→  const session = getSession(roomId);
  1136→  1136→  const cipher = getCipher(roomId);
  1137→  1137→  if (!session || !cipher) return;
  1138→  1138→
  1139→  1139→  const result = await cipher.open(frame);
  1140→  1140→  await handleOpenResult(roomId, frame, result);
  1141→  1141→
  1142→  1142→  // A frame we could not yet decrypt may mean we are a joiner (or a
  1143→  1143→  // reconnecter) waiting for key delivery — ask for it, quietly.
  1144→  1144→  if (result.type === "pending" && cipher.kv === 1) {
  1145→  1145→    const now = Date.now();
  1146→  1146→    const last = keyReqLastEmit.get(roomId) ?? 0;
  1147→  1147→    if (now - last > KEYREQ_THROTTLE_MS) {
  1148→  1148→      keyReqLastEmit.set(roomId, now);
  1149→  1149→      getRelay().emit("key:request", { roomId });
  1150→  1150→      setResealing(roomId, true);
  1151→  1151→    }
  1152→  1152→  }
  1153→  1153→}

  1155→  1155→async function handleOpenResult(roomId: string, frame: WireFrame, result: OpenResult) {
  1156→  1156→  const session = getSession(roomId);
  1157→  1157→  const cipher = getCipher(roomId);
  1158→  1158→  if (!session || !cipher) return;
  1159→  1159→  const registry = useApp.getState().members[roomId] ?? [];
  1160→  1160→  const inRoom = useApp.getState().activeRoomId === roomId;
  1161→  1161→
  1162→  1162→  switch (result.type) {
  1163→  1163→    case "text": {
  1164→  1164→      const sender = registry.find((m) => m.memberId === result.body.senderId);
  1165→  1165→      if (!sender) return;
  1166→  1166→      const ttlSec = result.body.ttlSec;
  1167→  1167→      const view: MessageView = {
  1168→  1168→        id: frame.id,
  1169→  1169→        kind: "text",
  1170→  1170→        senderId: sender.memberId,
  1171→  1171→        senderAlias: sender.alias,
  1172→  1172→        senderColor: sender.colorIdx,
  1173→  1173→        self: sender.memberId === session.memberId,
  1174→  1174→        status: "sent",
  1175→  1175→        ts: result.body.ts,
  1176→  1176→        text: result.body.text ?? "",
  1177→  1177→        ttlSec,
  1178→  1178→        expiresAt: ttlSec ? result.body.ts + ttlSec * 1000 : undefined,
  1179→  1179→      };
  1180→  1180→      useApp.setState((s) => ({
  1181→  1181→        messages: {
  1182→  1182→          ...s.messages,
  1183→  1183→          [roomId]: [...(s.messages[roomId] ?? []), view],
  1184→  1184→        },
  1185→  1185→      }));
  1186→  1186→      clearTyping(roomId, sender.memberId);
  1187→  1187→      touchCard(roomId, !inRoom);
  1188→  1188→      if (view.expiresAt) {
  1189→  1189→        scheduleBurn(roomId, view.id, view.expiresAt, onMessageBurn);
  1190→  1190→      }
  1191→  1191→      return;
  1192→  1192→    }
  1193→  1193→
  1194→  1194→    case "typing": {
  1195→  1195→      if (result.senderId === session.memberId) return;
  1196→  1196→      const sender = registry.find((m) => m.memberId === result.senderId);
  1197→  1197→      useApp.setState((s) => {
  1198→  1198→        const list = (s.typing[roomId] ?? []).filter((t) => t.memberId !== result.senderId);
  1199→  1199→        return {
  1200→  1200→          typing: {
  1201→  1201→            ...s.typing,
  1202→  1202→            [roomId]: [
  1203→  1203→              ...list,
  1204→  1204→              { memberId: result.senderId, alias: sender?.alias ?? "Someone", at: Date.now() },
  1205→  1205→            ],
  1206→  1206→          },
  1207→  1207→        };
  1208→  1208→      });
  1209→  1209→      pruneTypingLater(roomId);
  1210→  1210→      return;
  1211→  1211→    }
  1212→  1212→
  1213→  1213→    case "spent": {
  1214→  1214→      patchView(roomId, result.messageId, { spent: true });
  1215→  1215→      return;
  1216→  1216→    }
  1217→  1217→
  1218→  1218→    case "burn": {
  1219→  1219→      const target = (useApp.getState().messages[roomId] ?? []).find(
  1220→  1220→        (m) => m.id === result.messageId,
  1221→  1221→      );
  1222→  1222→      // The cipher already verified the author's signature; make sure
  1223→  1223→      // the target is actually the author's own letter.
  1224→  1224→      if (!target || target.kind === "system" || target.senderId !== result.senderId) return;
  1225→  1225→      const t = burnTimers.get(result.messageId);
  1226→  1226→      if (t) {
  1227→  1227→        clearTimeout(t);
  1228→  1228→        burnTimers.delete(result.messageId);
  1229→  1229→      }
  1230→  1230→      onMessageBurn(roomId, result.messageId);
  1231→  1231→      return;
  1232→  1232→    }
  1233→  1233→
  1234→  1234→    case "file": {
  1235→  1235→      const sender = registry.find((m) => m.memberId === result.senderId);
  1236→  1236→      if (!sender) return;
  1237→  1237→      const view: MessageView = {
  1238→  1238→        id: result.file.messageId,
  1239→  1239→        kind: "file",
  1240→  1240→        senderId: sender.memberId,
  1241→  1241→        senderAlias: sender.alias,
  1242→  1242→        senderColor: sender.colorIdx,
  1243→  1243→        self: sender.memberId === session.memberId,
  1244→  1244→        status: "sent",
  1245→  1245→        ts: result.ts,
  1246→  1246→        file: {
  1247→  1247→          name: result.file.name,
  1248→  1248→          mime: result.file.mime,
  1249→  1249→          size: result.file.size,
  1250→  1250→          dataB64: result.file.dataB64,
  1251→  1251→        },
  1252→  1252→        ttlSec: result.file.ttlSec,
  1253→  1253→        expiresAt: result.file.ttlSec ? result.ts + result.file.ttlSec * 1000 : undefined,
  1254→  1254→        viewOnce: result.file.viewOnce,
  1255→  1255→      };
  1256→  1256→      useApp.setState((s) => ({
  1257→  1257→        messages: {
  1258→  1258→          ...s.messages,
  1259→  1259→          [roomId]: [...(s.messages[roomId] ?? []), view],
  1260→  1260→        },
  1261→  1261→      }));
  1262→  1262→      clearTyping(roomId, sender.memberId);
  1263→  1263→      touchCard(roomId, !inRoom);
  1264→  1264→      if (view.expiresAt) {
  1265→  1265→        scheduleBurn(roomId, view.id, view.expiresAt, onMessageBurn);
  1266→  1266→      }
  1267→  1267→      return;
  1268→  1268→    }
  1269→  1269→
  1270→  1270→    case "file-meta":
  1271→  1271→    case "file-chunk":
  1272→  1272→      return; // assembly happens inside the cipher; render on completion
  1273→  1273→
  1274→  1274→    case "offer-installed": {
  1275→  1275→      session.kv = cipher.kv;
  1276→  1276→      setResealing(roomId, false);
  1277→  1277→      if (result.rotation) {
  1278→  1278→        addSystemLine(roomId, "The room re-sealed for those who remain.");
  1279→  1279→      }
  1280→  1280→      // Frames that were parked waiting for this key.
  1281→  1281→      const drained = await cipher.drainPending();
  1282→  1282→      for (const { frame: parked, result: r } of drained) {
  1283→  1283→        await handleOpenResult(roomId, parked, r).catch(() => undefined);
  1284→  1284→      }
  1285→  1285→      return;
  1286→  1286→    }
  1287→  1287→
  1288→  1288→    case "pending":
  1289→  1289→      return;
  1290→  1290→
  1291→  1291→    case "reject": {
  1292→  1292→      if (result.reason === "signature") {
  1293→  1293→        const sender = registry.find((m) => m.memberId === frame.from);
  1294→  1294→        addSystemLine(
  1295→  1295→          roomId,
  1296→  1296→          `A message claiming to be from ${sender?.alias ?? "an unknown member"} was rejected — signature invalid.`,
  1297→  1297→        );
  1298→  1298→        touchCard(roomId, !inRoom);
  1299→  1299→      }
  1300→  1300→      // Other rejections (replay, registry, kv, shape) are silent —
  1301→  1301→      // they are the protocol defending itself, not user-actionable.
  1302→  1302→      return;
  1303→  1303→    }
  1304→  1304→  }
  1305→  1305→}

  1311→  1311→async function receiveLegacyEnvelope(roomId: string, envelope: WireEnvelope) {
  1312→  1312→  const session = getSession(roomId);
  1313→  1313→  if (!session || !envelope) return;
  1314→  1314→
  1315→  1315→  let payload: Payload | null = null;
  1316→  1316→  try {
  1317→  1317→    const { getEpochKeyLegacy } = await import("@/lib/legacy");
  1318→  1318→    const key = await getEpochKeyLegacy(roomId, envelope.epoch, session.password);
  1319→  1319→    payload = await decryptJson<Payload>(key, envelope.iv, envelope.ct);
  1320→  1320→  } catch {
  1321→  1321→    payload = null;
  1322→  1322→  }
  1323→  1323→
  1324→  1324→  const registry = useApp.getState().members[roomId] ?? [];
  1325→  1325→  const sender = registry.find((m) => m.memberId === envelope.senderId);
  1326→  1326→
  1327→  1327→  if (!payload || !sender) {
  1328→  1328→    addSystemLine(
  1329→  1329→      roomId,
  1330→  1330→      `A message claiming to be from ${sender?.alias ?? "an unknown member"} was rejected — signature invalid.`,
  1331→  1331→    );
  1332→  1332→    touchCard(roomId, useApp.getState().activeRoomId !== roomId);
  1333→  1333→    return;
  1334→  1334→  }
  1335→  1335→
  1336→  1336→  const canonical = canonicalFor({
  1337→  1337→    roomId,
  1338→  1338→    epoch: envelope.epoch,
  1339→  1339→    senderId: envelope.senderId,
  1340→  1340→    ts: payload.ts,
  1341→  1341→    kind: envelope.kind,
  1342→  1342→    text: payload.text,
  1343→  1343→    fileSha: payload.file?.sha,
  1344→  1344→  });
  1345→  1345→  const ok = await verifyCanonical(sender.pubkey, canonical, payload.sig);
  1346→  1346→  if (!ok) {
  1347→  1347→    addSystemLine(
  1348→  1348→      roomId,
  1349→  1349→      `A message claiming to be from ${sender.alias} was rejected — signature invalid.`,
  1350→  1350→    );
  1351→  1351→    touchCard(roomId, useApp.getState().activeRoomId !== roomId);
  1352→  1352→    return;
  1353→  1353→  }
  1354→  1354→
  1355→  1355→  const view: MessageView = {
  1356→  1356→    id: envelope.id,
  1357→  1357→    kind: envelope.kind,
  1358→  1358→    senderId: sender.memberId,
  1359→  1359→    senderAlias: sender.alias,
  1360→  1360→    senderColor: sender.colorIdx,
  1361→  1361→    self: sender.memberId === session.memberId,
  1362→  1362→    status: "sent",
  1363→  1363→    ts: payload.ts,
  1364→  1364→    text: payload.text,
  1365→  1365→    file: payload.file
  1366→  1366→      ? {
  1367→  1367→          name: payload.file.name,
  1368→  1368→          mime: payload.file.mime,
  1369→  1369→          size: payload.file.size,
  1370→  1370→          dataB64: payload.file.dataB64,
  1371→  1371→        }
  1372→  1372→      : undefined,
  1373→  1373→    ttlSec: envelope.ttlSec,
  1374→  1374→    expiresAt: envelope.ttlSec ? payload.ts + envelope.ttlSec * 1000 : undefined,
  1375→  1375→    viewOnce: envelope.viewOnce,
  1376→  1376→  };
  1377→  1377→  useApp.setState((s) => ({
  1378→  1378→    messages: {
  1379→  1379→      ...s.messages,
  1380→  1380→      [roomId]: [...(s.messages[roomId] ?? []), view],
  1381→  1381→    },
  1382→  1382→  }));
  1383→  1383→  clearTyping(roomId, sender.memberId);
  1384→  1384→  touchCard(roomId, useApp.getState().activeRoomId !== roomId);
  1385→  1385→  if (view.expiresAt) {
  1386→  1386→    scheduleBurn(roomId, view.id, view.expiresAt, onMessageBurn);
  1387→  1387→  }
  1388→  1388→}
