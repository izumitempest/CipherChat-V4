   114→   114→/* ------------ TTL burn scheduling (module memory) ------------ */
   115→   115→const burnTimers = new Map<string, ReturnType<typeof setTimeout>>();
   116→   116→
   117→   117→/* ------------ typing whispers (transient by design) ------------ */
   118→   118→const TYPING_TTL_MS = 3500;
   119→   119→const TYPING_EMIT_THROTTLE_MS = 2500;
   120→   120→const typingLastEmit = new Map<string, number>();
   121→   121→const typingPruneTimers = new Map<string, ReturnType<typeof setTimeout>>();
   122→   122→
   123→   123→function pruneTypingLater(roomId: string) {
   124→   124→  const existing = typingPruneTimers.get(roomId);
   125→   125→  if (existing) clearTimeout(existing);
   126→   126→  const t = setTimeout(() => {
   127→   127→    typingPruneTimers.delete(roomId);
   128→   128→    useApp.setState((s) => {
   129→   129→      const fresh = (s.typing[roomId] ?? []).filter(
   130→   130→        (x) => Date.now() - x.at < TYPING_TTL_MS,
   131→   131→      );
   132→   132→      return { typing: { ...s.typing, [roomId]: fresh } };
   133→   133→    });
   134→   134→  }, TYPING_TTL_MS + 100);
   135→   135→  typingPruneTimers.set(roomId, t);
   136→   136→}
   137→   137→
   138→   138→function clearTyping(roomId: string, memberId: string) {
   139→   139→  useApp.setState((s) => {
   140→   140→    const list = s.typing[roomId] ?? [];
   141→   141→    if (!list.some((t) => t.memberId === memberId)) return {};
   142→   142→    return {
   143→   143→      typing: {
   144→   144→        ...s.typing,
   145→   145→        [roomId]: list.filter((t) => t.memberId !== memberId),
   146→   146→      },
   147→   147→    };
   148→   148→  });
   149→   149→}
   150→   150→
   151→   151→function scheduleBurn(
   152→   152→  roomId: string,
   153→   153→  messageId: string,
   154→   154→  expiresAt: number,
   155→   155→  onBurn: (roomId: string, messageId: string) => void,
   156→   156→) {
   157→   157→  if (burnTimers.has(messageId)) return;
   158→   158→  const delay = Math.max(0, expiresAt - Date.now());
   159→   159→  const t = setTimeout(() => {
   160→   160→    burnTimers.delete(messageId);
   161→   161→    onBurn(roomId, messageId);
   162→   162→  }, delay);
   163→   163→  burnTimers.set(messageId, t);
   164→   164→}
   165→   165→
   166→   166→function cancelBurns(roomId: string, messages: MessageView[]) {
   167→   167→  for (const m of messages) {
   168→   168→    if (m.id) {
   169→   169→      const t = burnTimers.get(m.id);
   170→   170→      if (t) {
   171→   171→        clearTimeout(t);
   172→   172→        burnTimers.delete(m.id);
   173→   173→      }
   174→   174→    }
   175→   175→  }
   176→   176→  void roomId;
   177→   177→}

   199→   199→  init: async () => {
   200→   200→    if (get().ready) return;

   212→   212→  syncHash: () => {
   213→   213→    const hash = window.location.hash.replace(/^#\/?/, "");
   214→   214→    if (hash.startsWith("join")) {
   215→   215→      const code = hash.split("/")[1] ?? "";
   216→   216→      set({ screen: "invite", inviteCode: code || null, activeRoomId: null });
   217→   217→      return;
   218→   218→    }
   219→   219→    if (hash.startsWith("new")) {
   220→   220→      set({ screen: "landing", activeRoomId: null });
   221→   221→      return;
   222→   222→    }
   223→   223→    if (hash.startsWith("rooms")) {
   224→   224→      set({ screen: "rooms", activeRoomId: null });
   225→   225→      return;
   226→   226→    }
   227→   227→    if (hash.startsWith("r/")) {
   228→   228→      const roomId = hash.slice(2);
   229→   229→      const session = getSession(roomId);
   230→   230→      set({ screen: "chat", activeRoomId: roomId });
   231→   231→      if (session) {
   232→   232→        patchRoomCard(roomId, { unread: false });
   233→   233→        set({ roomCards: loadRoomCards() });
   234→   234→      }
   235→   235→      // No keys in memory — the room is locked. A first-class state,
   236→   236→      // never an error.
   237→   237→      return;
   238→   238→    }
   239→   239→    // Root: the desk if you have rooms, the landing if you don't.
   240→   240→    if (get().roomCards.length > 0) {
   241→   241→      set({ screen: "rooms", activeRoomId: null });
   242→   242→    } else {
   243→   243→      set({ screen: "landing", activeRoomId: null });
   244→   244→    }
   245→   245→  },
   246→   246→
   247→   247→  navigate: (screen, roomId = null) => {
   248→   248→    const hash =
   249→   249→      screen === "rooms"
   250→   250→        ? "#/rooms"
   251→   251→        : screen === "chat" && roomId
   252→   252→          ? `#/r/${roomId}`
   253→   253→          : screen === "invite"
   254→   254→            ? "#/join"
   255→   255→            : screen === "landing"
   256→   256→              ? "#/new"
   257→   257→              : "#/";
   258→   258→    if (window.location.hash !== hash) {
   259→   259→      window.location.hash = hash;
   260→   260→    } else {
   261→   261→      get().syncHash();
   262→   262→    }
   263→   263→  },

   283→   283→      fetch(`/api/rooms/${roomId}/verifier`, {

   294→   294→      method: "POST",

   299→   299→      set({ sealing: null });

   314→   314→      pending: new Map(),
   315→   315→    };

   317→   317→    set({ sealing: null });

   320→   320→  },

   322→   322→  /* ------------------------------------------------- join ---- */
   323→   323→
   324→   324→  joinRoom: async (code, password, localName) => {
   325→   325→    const device = get().device;
   326→   326→    if (!device) return { ok: false, reason: "error" };
   327→   327→
   328→   328→    const infoRes = await fetch(`/api/rooms/${encodeURIComponent(code)}`);
   329→   329→    if (!infoRes.ok) {
   330→   330→      const info = await infoRes.json().catch(() => ({}));
   331→   331→      return {
   332→   332→        ok: false,
   333→   333→        reason: info?.burned ? "burned" : "not-found",
   334→   334→      };
   335→   335→    }
   336→   336→    const info = await infoRes.json();
   337→   337→    if (info.memberCount >= info.memberCap) {
   338→   338→      return { ok: false, reason: "room-full" };
   339→   339→    }
   340→   340→
   341→   341→    const { deriveRoomKey } = await import("@/lib/crypto");
   342→   342→    set({ sealing: { roomId: code, label: "Sealing the room" } });
   343→   343→    const started = Date.now();
   344→   344→    const key = await deriveRoomKey(code, info.epoch, password);
   345→   345→    const verified = info.verifier ? await checkVerifier(key, info.verifier) : true;
   346→   346→    await new Promise((r) =>
   347→   347→      setTimeout(r, Math.max(0, MIN_SEAL_MS - (Date.now() - started))),
   348→   348→    );
   349→   349→
   350→   350→    if (!verified) {
   351→   351→      set({ sealing: null });
   352→   352→      return { ok: false, reason: "wrong-password" };
   353→   353→    }
   354→   354→
   355→   355→    const alias = aliasFromFingerprint(device.fingerprintHex);
   356→   356→    const colorIdx = inkFromFingerprint(device.fingerprintHex);
   357→   357→    const joinRes = await fetch(`/api/rooms/${code}/members`, {
   358→   358→      method: "POST",
   359→   359→      headers: { "content-type": "application/json" },
   360→   360→      body: JSON.stringify({ pubkey: device.pubJwk, alias, colorIdx }),
   361→   361→    });
   362→   362→    if (!joinRes.ok) {
   363→   363→      set({ sealing: null });
   364→   364→      if (joinRes.status === 403) return { ok: false, reason: "room-full" };
   365→   365→      return { ok: false, reason: "error" };
   366→   366→    }
   367→   367→    const { memberId, epoch, rejoined } = await joinRes.json();
   368→   368→
   369→   369→    const session: RoomSession = {
   370→   370→      roomId: code,
   371→   371→      memberId,
   372→   372→      alias,
   373→   373→      colorIdx,
   374→   374→      epoch,
   375→   375→      password,
   376→   376→      creatorToken: loadCreatorToken(code),
   377→   377→      defaultTtl: loadRoomSettings(code).defaultTtl,
   378→   378→      keys: new Map([[epoch, key]]),
   379→   379→      pending: new Map(),
   380→   380→    };
   381→   381→    setSession(session);
   382→   382→    set({ sealing: null });
   383→   383→    const name =
   384→   384→      localName ||
   385→   385→      get().roomCards.find((c) => c.roomId === code)?.localName ||
   386→   386→      `Room ${code.slice(0, 4)}`;
   387→   387→    await get().enterRoom(session, name, !!rejoined);
   388→   388→    return { ok: true };
   389→   389→  },

   393→   393→  enterRoom: async (session, localName, rejoined) => {
   394→   394→    const now = Date.now();
   395→   395→    upsertRoomCard({
   396→   396→      roomId: session.roomId,
   397→   397→      localName,
   398→   398→      createdAt: now,
   399→   399→      lastActivity: now,
   400→   400→      lastMembers: 1,
   401→   401→      unread: false,
   402→   402→      burned: false,
   403→   403→      locked: false,
   404→   404→    });
   405→   405→    set({ roomCards: loadRoomCards() });
   406→   406→
   407→   407→    // Fresh message memory — there is no backlog, ever.
   408→   408→    set((s) => ({ messages: { ...s.messages, [session.roomId]: [] } }));
   409→   409→
   410→   410→    await get().refreshMembers(session.roomId);
   411→   411→
   412→   412→    const device = get().device;
   413→   413→    const relay = getRelay();
   414→   414→    relay.emit("room:join", {
   415→   415→      roomId: session.roomId,
   416→   416→      memberId: session.memberId,
   417→   417→      alias: session.alias,
   418→   418→      colorIdx: session.colorIdx,
   419→   419→      pubkey: device?.pubJwk,
   420→   420→      rejoined,
   421→   421→    });
   422→   422→    set({ relayOnline: relay.connected });
   423→   423→
   424→   424→    get().navigate("chat", session.roomId);
   425→   425→  },
   426→   426→
   427→   427→  refreshMembers: async (roomId) => {
   428→   428→    try {
   429→   429→      const res = await fetch(`/api/rooms/${roomId}/members`);
   430→   430→      if (!res.ok) return;
   431→   431→      const { members } = await res.json();
   432→   432→      set((s) => ({ members: { ...s.members, [roomId]: members } }));
   433→   433→    } catch {
   434→   434→      /* offline — the registry stays as-is */
   435→   435→    }
   436→   436→  },

   438→   438→  /* ------------------------------------------------- send ---- */
   439→   439→
   440→   440→  sendMessage: async (text, file, ttlOverride) => {
   441→   441→    const viewOnce = !!file?.viewOnce;
   442→   442→    const { activeRoomId, device } = get();
   443→   443→    if (!activeRoomId || !device) return;
   444→   444→    const session = getSession(activeRoomId);
   445→   445→    if (!session) return;
   446→   446→    const trimmed = text.trim();
   447→   447→    if (!trimmed && !file) return;
   448→   448→
   449→   449→    const id = crypto.randomUUID();
   450→   450→    const ts = Date.now();
   451→   451→    const kind = file ? "file" : "text";
   452→   452→    let fileSha: string | undefined;
   453→   453→    let fullFile: FilePayload | undefined;
   454→   454→    if (file) {
   455→   455→      fileSha = await sha256HexOfBytes(fromB64(file.dataB64));
   456→   456→      fullFile = { ...file, sha: fileSha };
   457→   457→    }
   458→   458→
   459→   459→    const canonical = canonicalFor({
   460→   460→      roomId: session.roomId,
   461→   461→      epoch: session.epoch,
   462→   462→      senderId: session.memberId,
   463→   463→      ts,
   464→   464→      kind,
   465→   465→      text: trimmed,
   466→   466→      fileSha,
   467→   467→    });
   468→   468→    const sig = await signCanonical(device.privJwk, canonical);
   469→   469→    const payload: Payload = {
   470→   470→      text: trimmed,
   471→   471→      ts,
   472→   472→      senderId: session.memberId,
   473→   473→      sig,
   474→   474→      kind,
   475→   475→      file: fullFile,
   476→   476→    };
   477→   477→
   478→   478→    const key = await getEpochKey(session, session.epoch);
   479→   479→    const { iv, ct } = await encryptJson(key, payload);
   480→   480→
   481→   481→    const ttlSec = (ttlOverride !== undefined ? ttlOverride : session.defaultTtl) || undefined;
   482→   482→    const envelope: WireEnvelope = {
   483→   483→      id,
   484→   484→      roomId: session.roomId,
   485→   485→      epoch: session.epoch,
   486→   486→      senderId: session.memberId,
   487→   487→      ts,
   488→   488→      iv,
   489→   489→      ct,
   490→   490→      ttlSec,
   491→   491→      viewOnce: file && viewOnce ? true : undefined,
   492→   492→      kind,
   493→   493→    };
   494→   494→
   495→   495→    const view: MessageView = {
   496→   496→      id,
   497→   497→      kind,
   498→   498→      senderId: session.memberId,
   499→   499→      senderAlias: session.alias,
   500→   500→      senderColor: session.colorIdx,
   501→   501→      self: true,
   502→   502→      status: "sending",
   503→   503→      ts,
   504→   504→      text: trimmed,
   505→   505→      file: file
   506→   506→        ? { name: file.name, mime: file.mime, size: file.size, dataB64: file.dataB64 }
   507→   507→        : undefined,
   508→   508→      ttlSec,
   509→   509→      expiresAt: ttlSec ? ts + ttlSec * 1000 : undefined,
   510→   510→      viewOnce: file && viewOnce ? true : undefined,
   511→   511→    };
   512→   512→
   513→   513→    set((s) => ({
   514→   514→      messages: {
   515→   515→        ...s.messages,
   516→   516→        [session.roomId]: [...(s.messages[session.roomId] ?? []), view],
   517→   517→      },
   518→   518→    }));
   519→   519→    patchRoomCard(session.roomId, { lastActivity: ts });
   520→   520→    set({ roomCards: loadRoomCards() });
   521→   521→
   522→   522→    if (ttlSec) {
   523→   523→      scheduleBurn(session.roomId, id, ts + ttlSec * 1000, onMessageBurn);
   524→   524→    }
   525→   525→
   526→   526→    getRelay().emit("message:send", { roomId: session.roomId, envelope });
   527→   527→  },
   528→   528→
   529→   529→  /* --------------------------------------------- typing ------- */
   530→   530→
   531→   531→  emitTyping: (roomId) => {
   532→   532→    const session = getSession(roomId);
   533→   533→    if (!session) return;
   534→   534→    const now = Date.now();
   535→   535→    const last = typingLastEmit.get(roomId) ?? 0;
   536→   536→    if (now - last < TYPING_EMIT_THROTTLE_MS) return;
   537→   537→    typingLastEmit.set(roomId, now);
   538→   538→    getRelay().emit("member:typing", {
   539→   539→      roomId,
   540→   540→      memberId: session.memberId,
   541→   541→      alias: session.alias,
   542→   542→    });
   543→   543→  },
   544→   544→
   545→   545→  /* --------------------------------------------- view-once --- */
   546→   546→
   547→   547→  spendViewOnce: (roomId, messageId) => {
   548→   548→    set((s) => {
   549→   549→      const list = s.messages[roomId] ?? [];
   550→   550→      return {
   551→   551→        messages: {
   552→   552→          ...s.messages,
   553→   553→          [roomId]: list.map((m) => (m.id === messageId ? { ...m, spent: true } : m)),
   554→   554→        },
   555→   555→      };
   556→   556→    });
   557→   557→    getRelay().emit("message:spent", { roomId, messageId });
   558→   558→  },
   559→   559→
   560→   560→  /* --------------------------------------------- burn-one ---- */
   561→   561→
   562→   562→  /** Retire one of your own letters ahead of its clock. The announce
   563→   563→   *  is signed, so only the author can do this — nobody else can
   564→   564→   *  burn a message they did not write. */
   565→   565→  burnMessage: async (roomId, messageId) => {
   566→   566→    const { activeRoomId, device } = get();
   567→   567→    if (!device) return;
   568→   568→    const target = roomId ?? activeRoomId;
   569→   569→    const session = getSession(target);
   570→   570→    const msg = (get().messages[target] ?? []).find((m) => m.id === messageId);
   571→   571→    if (!session || !msg || !msg.self || msg.kind === "system") return;
   572→   572→
   573→   573→    const sig = await signCanonical(
   574→   574→      device.privJwk,
   575→   575→      burnCanonical(target, session.memberId, messageId),
   576→   576→    );
   577→   577→    getRelay().emit("message:burn", {
   578→   578→      roomId: target,
   579→   579→      messageId,
   580→   580→      senderId: session.memberId,
   581→   581→      sig,
   582→   582→    });
   583→   583→
   584→   584→    const t = burnTimers.get(messageId);
   585→   585→    if (t) {
   586→   586→      clearTimeout(t);
   587→   587→      burnTimers.delete(messageId);
   588→   588→    }
   589→   589→    onMessageBurn(target, messageId);
   590→   590→  },
   591→   591→
   592→   592→  /* ------------------------------------------------- burn ---- */
   593→   593→
   594→   594→  burnRoom: async (roomId) => {
   595→   595→    const session = getSession(roomId);
   596→   596→    if (!session?.creatorToken) return;
   597→   597→    try {
   598→   598→      await fetch(`/api/rooms/${roomId}/burn`, {
   599→   599→        method: "POST",
   600→   600→        headers: { "content-type": "application/json" },
   601→   601→        body: JSON.stringify({ creatorToken: session.creatorToken }),
   602→   602→      });
   603→   603→    } catch {
   604→   604→      /* even if this fails, members burn on the announce */
   605→   605→    }
   606→   606→    getRelay().emit("room:burn", { roomId });
   607→   607→    set({ burn: { roomId } });
   608→   608→  },
   609→   609→
   610→   610→  finishRoomBurn: (roomId) => {
   611→   611→    const messages = get().messages[roomId] ?? [];
   612→   612→    cancelBurns(roomId, messages);
   613→   613→    dropSession(roomId);
   614→   614→    patchRoomCard(roomId, { burned: true, unread: false, lastActivity: Date.now() });
   615→   615→    set((s) => ({
   616→   616→      burn: null,
   617→   617→      roomCards: loadRoomCards(),
   618→   618→      activeRoomId: null,
   619→   619→      messages: { ...s.messages, [roomId]: [] },
   620→   620→    }));
   621→   621→    get().navigate("rooms");
   622→   622→  },
   623→   623→
   624→   624→  /* ------------------------------------------------- leave --- */
   625→   625→
   626→   626→  leaveRoom: async (roomId) => {
   627→   627→    const session = getSession(roomId);
   628→   628→    if (session) {
   629→   629→      try {
   630→   630→        await fetch(`/api/rooms/${roomId}/leave`, {
   631→   631→          method: "POST",
   632→   632→          headers: { "content-type": "application/json" },
   633→   633→          body: JSON.stringify({ memberId: session.memberId }),
   634→   634→        });
   635→   635→      } catch {
   636→   636→        /* best effort — the epoch bump happens server-side anyway */
   637→   637→      }
   638→   638→      getRelay().emit("member:leave", {
   639→   639→        roomId,
   640→   640→        memberId: session.memberId,
   641→   641→        alias: session.alias,
   642→   642→      });
   643→   643→    }
   644→   644→    const messages = get().messages[roomId] ?? [];
   645→   645→    cancelBurns(roomId, messages);
   646→   646→    dropSession(roomId);
   647→   647→    removeRoomCard(roomId);
   648→   648→    set((s) => ({
   649→   649→      roomCards: loadRoomCards(),
   650→   650→      activeRoomId: null,
   651→   651→      messages: { ...s.messages, [roomId]: [] },
   652→   652→    }));
   653→   653→    get().navigate("rooms");
   654→   654→  },
   655→   655→
   656→   656→  /* ---------------------------------------------- settings --- */
   657→   657→
   658→   658→  renameRoom: (roomId, name) => {
   659→   659→    patchRoomCard(roomId, { localName: name });
   660→   660→    set({ roomCards: loadRoomCards() });
   661→   661→  },
   662→   662→
   663→   663→  setDefaultTtl: (roomId, ttl) => {
   664→   664→    const session = getSession(roomId);
   665→   665→    if (session) session.defaultTtl = ttl;
   666→   666→    saveRoomSettings(roomId, { defaultTtl: ttl });
   667→   667→  },
   668→   668→
   669→   669→  markVerified: (roomId, memberId, verified) => {
   670→   670→    const current = loadVerified(roomId);
   671→   671→    const next = verified
   672→   672→      ? Array.from(new Set([...current, memberId]))
   673→   673→      : current.filter((id) => id !== memberId);
   674→   674→    saveVerified(roomId, next);
   675→   675→    set((s) => ({ members: { ...s.members } })); // trigger re-render
   676→   676→  },
   677→   677→
   678→   678→  isVerified: (roomId, memberId) => loadVerified(roomId).includes(memberId),
   679→   679→}));

   685→   685→function onMessageBurn(roomId: string, messageId: string) {
   686→   686→  // Phase 1: burning (CSS animation 600ms)
   687→   687→  useApp.setState((s) => ({
   688→   688→    messages: {
   689→   689→      ...s.messages,
   690→   690→      [roomId]: (s.messages[roomId] ?? []).map((m) =>
   691→   691→        m.id === messageId ? { ...m, status: "burning" } : m,
   692→   692→      ),
   693→   693→    },
   694→   694→  }));
   695→   695→  // Phase 2: gone.
   696→   696→  setTimeout(() => {
   697→   697→    useApp.setState((s) => ({
   698→   698→      messages: {
   699→   699→        ...s.messages,
   700→   700→        [roomId]: (s.messages[roomId] ?? []).filter((m) => m.id !== messageId),
   701→   701→      },
   702→   702→    }));
   703→   703→  }, 620);
   704→   704→}

   738→   738→function wireRelay(
   739→   739→  set: (
   740→   740→    partial:
   741→   741→      | Partial<AppState>
   742→   742→      | ((state: AppState) => Partial<AppState>),
   743→   743→  ) => void,
   744→   744→  get: () => AppState,
   745→   745→) {
   746→   746→  const relay = getRelay();
   747→   747→
   748→   748→  relay.on("connect", () => {
   749→   749→    set({ relayOnline: true });
   750→   750→    const device = useApp.getState().device;
   751→   751→    for (const roomId of sessionRoomIds()) {
   752→   752→      const session = getSession(roomId);
   753→   753→      if (session) {
   754→   754→        relay.emit("room:join", {
   755→   755→          roomId,
   756→   756→          memberId: session.memberId,
   757→   757→          alias: session.alias,
   758→   758→          colorIdx: session.colorIdx,
   759→   759→          pubkey: device?.pubJwk,
   760→   760→          rejoined: true,
   761→   761→        });
   762→   762→      }
   763→   763→    }
   764→   764→  });
   765→   765→
   766→   766→  relay.on("disconnect", () => set({ relayOnline: false }));
   767→   767→
   768→   768→  relay.on("room:state", ({ roomId, members }: { roomId: string; members: MemberPublic[] }) => {
   769→   769→    // merge live presence into the registry
   770→   770→    const registry = useApp.getState().members[roomId] ?? [];
   771→   771→    const byId = new Map(registry.map((m) => [m.memberId, { ...m }]));
   772→   772→    for (const live of members) {
   773→   773→      const existing = byId.get(live.memberId);
   774→   774→      byId.set(live.memberId, {
   775→   775→        ...live,
   776→   776→        ...(existing?.pubkey ? { pubkey: existing.pubkey } : {}),
   777→   777→      });
   778→   778→    }
   779→   779→    set((s) => ({ members: { ...s.members, [roomId]: Array.from(byId.values()) } }));
   780→   780→  });
   781→   781→
   782→   782→  relay.on(
   783→   783→    "member:joined",
   784→   784→    ({ roomId, member, rejoined }: { roomId: string; member: MemberPublic; rejoined?: boolean }) => {
   785→   785→      const session = getSession(roomId);
   786→   786→      if (!session) return;
   787→   787→      set((s) => {
   788→   788→        const registry = s.members[roomId] ?? [];
   789→   789→        if (registry.some((m) => m.memberId === member.memberId)) {
   790→   790→          return {
   791→   791→            members: {
   792→   792→              ...s.members,
   793→   793→              [roomId]: registry.map((m) =>
   794→   794→                m.memberId === member.memberId ? { ...m, connected: true } : m,
   795→   795→              ),
   796→   796→            },
   797→   797→          };
   798→   798→        }
   799→   799→        return {
   800→   800→          members: { ...s.members, [roomId]: [...registry, { ...member, connected: true }] },
   801→   801→        };
   802→   802→      });
   803→   803→      if (member.memberId !== session.memberId) {
   804→   804→        addSystemLine(roomId, rejoined ? `${member.alias} returned.` : `${member.alias} joined.`);
   805→   805→        touchCard(roomId, get().activeRoomId !== roomId);
   806→   806→      }
   807→   807→    },
   808→   808→  );
   809→   809→
   810→   810→  relay.on(
   811→   811→    "member:presence",
   812→   812→    ({ roomId, memberId, connected }: { roomId: string; memberId: string; connected: boolean }) => {
   813→   813→      set((s) => ({
   814→   814→        members: {
   815→   815→          ...s.members,
   816→   816→          [roomId]: (s.members[roomId] ?? []).map((m) =>
   817→   817→            m.memberId === memberId ? { ...m, connected } : m,
   818→   818→          ),
   819→   819→        },
   820→   820→      }));
   821→   821→    },
   822→   822→  );
   823→   823→
   824→   824→  relay.on(
   825→   825→    "member:left",
   826→   826→    async ({ roomId, alias }: { roomId: string; memberId: string; alias: string }) => {
   827→   827→      const session = getSession(roomId);
   828→   828→      if (!session) return;
   829→   829→      addSystemLine(roomId, `${alias} left. The room key was rotated.`);
   830→   830→      // Rotate: fetch the new epoch and re-derive quietly.
   831→   831→      set((s) => ({ resealing: { ...s.resealing, [roomId]: true } }));
   832→   832→      try {
   833→   833→        const res = await fetch(`/api/rooms/${roomId}`);
   834→   834→        if (res.ok) {
   835→   835→          const info = await res.json();
   836→   836→          session.epoch = info.epoch;
   837→   837→          await getEpochKey(session, info.epoch);
   838→   838→        }
   839→   839→      } finally {
   840→   840→        set((s) => {
   841→   841→          const next = { ...s.resealing };
   842→   842→          delete next[roomId];
   843→   843→          return { resealing: next };
   844→   844→        });
   845→   845→      }
   846→   846→      get().refreshMembers(roomId);
   847→   847→    },
   848→   848→  );
   849→   849→
   850→   850→  relay.on("message:ack", ({ id, roomId }: { id: string; roomId?: string }) => {
   851→   851→    const target = roomId ?? get().activeRoomId;
   852→   852→    if (!target) return;
   853→   853→    patchView(target, id, { status: "sent" });
   854→   854→  });
   855→   855→
   856→   856→  relay.on("message:spent", ({ roomId, messageId }: { roomId: string; messageId: string }) => {
   857→   857→    patchView(roomId, messageId, { spent: true });
   858→   858→  });
   859→   859→
   860→   860→  relay.on(
   861→   861→    "message:burn",
   862→   862→    async ({
   863→   863→      roomId,
   864→   864→      messageId,
   865→   865→      senderId,
   866→   866→      sig,
   867→   867→    }: {
   868→   868→      roomId: string;
   869→   869→      messageId: string;
   870→   870→      senderId?: string;
   871→   871→      sig?: string;
   872→   872→    }) => {
   873→   873→      const session = getSession(roomId);
   874→   874→      if (!session || !senderId) return;
   875→   875→      const target = (useApp.getState().messages[roomId] ?? []).find(
   876→   876→        (m) => m.id === messageId,
   877→   877→      );
   878→   878→      if (!target || target.kind === "system") return;
   879→   879→
   880→   880→      const registry = useApp.getState().members[roomId] ?? [];
   881→   881→      const author = registry.find((m) => m.memberId === senderId);
   882→   882→      const ok =
   883→   883→        !!author?.pubkey &&
   884→   884→        !!sig &&
   885→   885→        target.senderId === senderId &&
   886→   886→        (await verifyCanonical(author.pubkey, burnCanonical(roomId, senderId, messageId), sig));
   887→   887→      if (!ok) {
   888→   888→        addSystemLine(
   889→   889→          roomId,
   890→   890→          `A burn request claiming to be from ${author?.alias ?? "an unknown member"} was rejected — signature invalid.`,
   891→   891→        );
   892→   892→        return;
   893→   893→      }
   894→   894→      const t = burnTimers.get(messageId);
   895→   895→      if (t) {
   896→   896→        clearTimeout(t);
   897→   897→        burnTimers.delete(messageId);
   898→   898→      }
   899→   899→      onMessageBurn(roomId, messageId);
   900→   900→    },
   901→   901→  );
   902→   902→
   903→   903→  relay.on("room:burned", ({ roomId }: { roomId: string }) => {
   904→   904→    const session = getSession(roomId);
   905→   905→    if (!session) return;
   906→   906→    set({ burn: { roomId } });
   907→   907→  });
   908→   908→
   909→   909→  relay.on(
   910→   910→    "message:new",
   911→   911→    async ({ roomId, envelope }: { roomId: string; envelope: WireEnvelope }) => {
   912→   912→      const session = getSession(roomId);
   913→   913→      if (!session || !envelope) return;
   914→   914→
   915→   915→      // Might need a key for a newer epoch (quiet re-seal)
   916→   916→      set((s) => ({ resealing: { ...s.resealing, [roomId]: true } }));
   917→   917→      let payload: Payload | null = null;
   918→   918→      try {
   919→   919→        const key = await getEpochKey(session, envelope.epoch);
   920→   920→        payload = await decryptJson<Payload>(key, envelope.iv, envelope.ct);
   921→   921→      } catch {
   922→   922→        payload = null;
   923→   923→      } finally {
   924→   924→        set((s) => {
   925→   925→          const next = { ...s.resealing };
   926→   926→          delete next[roomId];
   927→   927→          return { resealing: next };
   928→   928→        });
   929→   929→      }
   930→   930→
   931→   931→      const registry = get().members[roomId] ?? [];
   932→   932→      const sender = registry.find((m) => m.memberId === envelope.senderId);
   933→   933→
   934→   934→      if (!payload || !sender) {
   935→   935→        // Tampered, undecryptable, or from an unknown member: rejected.
   936→   936→        addSystemLine(
   937→   937→          roomId,
   938→   938→          `A message claiming to be from ${sender?.alias ?? "an unknown member"} was rejected — signature invalid.`,
   939→   939→        );
   940→   940→        touchCard(roomId, get().activeRoomId !== roomId);
   941→   941→        return;
   942→   942→      }
   943→   943→
   944→   944→      const canonical = canonicalFor({
   945→   945→        roomId,
   946→   946→        epoch: envelope.epoch,
   947→   947→        senderId: envelope.senderId,
   948→   948→        ts: payload.ts,
   949→   949→        kind: envelope.kind,
   950→   950→        text: payload.text,
   951→   951→        fileSha: payload.file?.sha,
   952→   952→      });
   953→   953→      const ok = await verifyCanonical(sender.pubkey, canonical, payload.sig);
   954→   954→      if (!ok) {
   955→   955→        addSystemLine(
   956→   956→          roomId,
   957→   957→          `A message claiming to be from ${sender.alias} was rejected — signature invalid.`,
   958→   958→        );
   959→   959→        touchCard(roomId, get().activeRoomId !== roomId);
   960→   960→        return;
   961→   961→      }
   962→   962→
   963→   963→      if (envelope.epoch > session.epoch) session.epoch = envelope.epoch;
   964→   964→
   965→   965→      const view: MessageView = {
   966→   966→        id: envelope.id,
   967→   967→        kind: envelope.kind,
   968→   968→        senderId: sender.memberId,
   969→   969→        senderAlias: sender.alias,
   970→   970→        senderColor: sender.colorIdx,
   971→   971→        self: sender.memberId === session.memberId,
   972→   972→        status: "sent",
   973→   973→        ts: payload.ts,
   974→   974→        text: payload.text,
   975→   975→        file: payload.file
   976→   976→          ? {
   977→   977→              name: payload.file.name,
   978→   978→              mime: payload.file.mime,
   979→   979→              size: payload.file.size,
   980→   980→              dataB64: payload.file.dataB64,
   981→   981→            }
   982→   982→          : undefined,
   983→   983→        ttlSec: envelope.ttlSec,
   984→   984→        expiresAt: envelope.ttlSec ? payload.ts + envelope.ttlSec * 1000 : undefined,
