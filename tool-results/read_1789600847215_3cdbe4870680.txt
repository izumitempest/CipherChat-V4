     1→// The application store. One place owns: the desk (room cards),
     2→// per-room message memory, member registries, the relay wiring, and
     3→// the choreography of enter / unlock / send / burn.
     4→//
     5→// SECURITY MODEL (protocol v2 — see lib/protocol.ts + lib/room-protocol.ts):
     6→//   Every frame the relay carries is padded to a uniform size and
     7→//   encrypted; typing, receipts, burns, key offers and file chunks all
     8→//   ride the same indistinguishable frames. Replay protection, the
     9→//   member-registry eviction gate, signature checks and the ECDH key
    10→//   rotation ceremony all live in RoomCipher — this store is a thin
    11→//   adapter that turns OpenResults into UI.

   138→function pruneTypingLater(roomId: string) {
   139→  const existing = typingPruneTimers.get(roomId);
   140→  if (existing) clearTimeout(existing);
   141→  const t = setTimeout(() => {
   142→    typingPruneTimers.delete(roomId);
   143→    useApp.setState((s) => {
   144→      const fresh = (s.typing[roomId] ?? []).filter(
   145→        (x) => Date.now() - x.at < TYPING_TTL_MS,
   146→      );
   147→      return { typing: { ...s.typing, [roomId]: fresh } };
   148→    });
   149→  }, TYPING_TTL_MS + 100);
   150→  typingPruneTimers.set(roomId, t);
   151→}
   152→
   153→function clearTyping(roomId: string, memberId: string) {
   154→  useApp.setState((s) => {
   155→    const list = s.typing[roomId] ?? [];
   156→    if (!list.some((t) => t.memberId === memberId)) return {};
   157→    return {
   158→      typing: {
   159→        ...s.typing,
   160→        [roomId]: list.filter((t) => t.memberId !== memberId),
   161→      },
   162→    };
   163→  });
   164→}

   166→function scheduleBurn(
   167→  roomId: string,
   168→  messageId: string,
   169→  expiresAt: number,
   170→  onBurn: (roomId: string, messageId: string) => void,
   171→) {
   172→  if (burnTimers.has(messageId)) return;
   173→  const delay = Math.max(0, expiresAt - Date.now());
   174→  const t = setTimeout(() => {
   175→    burnTimers.delete(messageId);
   176→    onBurn(roomId, messageId);
   177→  }, delay);
   178→  burnTimers.set(messageId, t);
   179→}
   180→
   181→function cancelBurns(roomId: string, messages: MessageView[]) {
   182→  for (const m of messages) {
   183→    if (m.id) {
   184→      const t = burnTimers.get(m.id);
   185→      if (t) {
   186→        clearTimeout(t);
   187→        burnTimers.delete(m.id);
   188→      }
   189→    }
   190→  }
   191→  void roomId;
   192→}

   214→  init: async () => {
   215→    if (get().ready) return;
   216→    // One random seed per device; every room derives its own signing
   217→    // key from it, so registries can never be correlated across rooms.
   218→    const seed = await loadDeviceSeed();
   219→    // Burned rooms showed their ash last session; now they are gone.
   220→    const cards = sweepBurnedRooms();
   221→    set({ seed, roomCards: cards, ready: true });
   222→
   223→    get().syncHash();
   224→    window.addEventListener("hashchange", () => get().syncHash());
   225→
   226→    wireRelay(set, get);

   229→  syncHash: () => {
   230→    const hash = window.location.hash.replace(/^#\/?/, "");
   231→    if (hash.startsWith("join")) {
   232→      const code = hash.split("/")[1] ?? "";
   233→      set({ screen: "invite", inviteCode: code || null, activeRoomId: null });
   234→      return;
   235→    }
   236→    if (hash.startsWith("new")) {
   237→      set({ screen: "landing", activeRoomId: null });
   238→      return;
   239→    }
   240→    if (hash.startsWith("rooms")) {
   241→      set({ screen: "rooms", activeRoomId: null });
   242→      return;
   243→    }
   244→    if (hash.startsWith("r/")) {
   245→      const roomId = hash.slice(2);
   246→      const session = getSession(roomId);
   247→      set({ screen: "chat", activeRoomId: roomId });
   248→      if (session) {
   249→        patchRoomCard(roomId, { unread: false });
   250→        set({ roomCards: loadRoomCards() });
   251→      }
   252→      // No keys in memory — the room is locked. A first-class state,
   253→      // never an error.
   254→      return;
   255→    }
   256→    // Root: the desk if you have rooms, the landing if you don't.
   257→    if (get().roomCards.length > 0) {
   258→      set({ screen: "rooms", activeRoomId: null });
   259→    } else {
   260→      set({ screen: "landing", activeRoomId: null });
   261→    }
   262→  },

   269→          ? `#/r/${roomId}`
   270→          : screen === "invite"
   271→            ? "#/join"
   272→            : screen === "landing"

   274→              : "#/";

   284→  createRoom: async (localName, password) => {
   285→    const seed = get().seed;
   286→    if (!seed) return { ok: false, reason: "error" };
   287→
   288→    const res = await fetch("/api/rooms", { method: "POST" });
   289→    if (!res.ok) return { ok: false, reason: "error" };
   290→    const { roomId, creatorToken } = await res.json();
   291→    saveCreatorToken(roomId, creatorToken);
   292→
   293→    set({ sealing: { roomId, label: "Sealing the room" } });
   294→    const started = Date.now();
   295→
   296→    // This room's own signing identity + session ECDH pair, and the
   297→    // argon2id key bundle (random salt, versioned, memory-hard).
   298→    const [identity, ecdh, bundle] = await Promise.all([
   299→      deriveRoomSigningKey(seed, roomId),
   300→      generateSessionEcdh(),
   301→      createKeyBundleV2(password),
   302→    ]);
   303→
   304→    await Promise.all([
   305→      fetch(`/api/rooms/${roomId}/verifier`, {
   306→        method: "PUT",
   307→        headers: { "content-type": "application/json" },
   308→        body: JSON.stringify({ verifier: JSON.stringify(bundle.bundle) }),
   309→      }),
   310→      new Promise((r) => setTimeout(r, MIN_SEAL_MS - (Date.now() - started))),
   311→    ]);
   312→
   313→    const alias = aliasFromFingerprint(identity.fingerprintHex);
   314→    const colorIdx = inkFromFingerprint(identity.fingerprintHex);
   315→    const joinRes = await fetch(`/api/rooms/${roomId}/members`, {
   316→      method: "POST",
   317→      headers: { "content-type": "application/json" },
   318→      body: JSON.stringify({
   319→        pubkey: identity.pubJwk,
   320→        ecdhPub: ecdh.pubRawB64,
   321→        alias,
   322→        colorIdx,
   323→      }),
   324→    });
   325→    if (!joinRes.ok) {
   326→      set({ sealing: null });
   327→      return { ok: false, reason: "error" };
   328→    }
   329→    const { memberId } = await joinRes.json();
   330→
   331→    const cipher = new RoomCipher({
   332→      roomId,
   333→      selfId: memberId,
   334→      sig: { privJwk: identity.privJwk, pubJwk: identity.pubJwk },
   335→      ecdh,
   336→      entryKey: bundle.key,
   337→    });
   338→    setCipher(roomId, cipher);
   339→
   340→    const session: RoomSession = {
   341→      roomId,
   342→      memberId,
   343→      alias,
   344→      colorIdx,
   345→      kv: 1,
   346→      password,
   347→      creatorToken,
   348→      defaultTtl: loadRoomSettings(roomId).defaultTtl,
   349→    };

   351→    set({ sealing: null });
   352→    await get().enterRoom(session, localName || "New room", false);
   353→    return { ok: true };
   354→  },

   356→  /* ------------------------------------------------- join ---- */
   357→
   358→  joinRoom: async (code, password, localName) => {
   359→    const seed = get().seed;
   360→    if (!seed) return { ok: false, reason: "error" };
   361→
   362→    const infoRes = await fetch(`/api/rooms/${encodeURIComponent(code)}`);
   363→    if (!infoRes.ok) {
   364→      const info = await infoRes.json().catch(() => ({}));
   365→      return {
   366→        ok: false,
   367→        reason: info?.burned ? "burned" : "not-found",
   368→      };
   369→    }
   370→    const info = await infoRes.json();
   371→    if (info.memberCount >= info.memberCap) {
   372→      return { ok: false, reason: "room-full" };
   373→    }
   374→
   375→    set({ sealing: { roomId: code, label: "Sealing the room" } });
   376→    const started = Date.now();
   377→    // The entry key (version 1) — argon2id for new rooms, PBKDF2 for
   378→    // rooms created before the upgrade. The bundle decides.
   379→    const entryKey = await unlockWithBundle(password, info.verifier, code, info.epoch);
   380→    await new Promise((r) =>
   381→      setTimeout(r, Math.max(0, MIN_SEAL_MS - (Date.now() - started))),
   382→    );
   383→
   384→    if (!entryKey) {
   385→      set({ sealing: null });
   386→      return { ok: false, reason: "wrong-password" };
   387→    }
   388→
   389→    const [identity, ecdh] = await Promise.all([
   390→      deriveRoomSigningKey(seed, code),
   391→      generateSessionEcdh(),
   392→    ]);
   393→    const alias = aliasFromFingerprint(identity.fingerprintHex);
   394→    const colorIdx = inkFromFingerprint(identity.fingerprintHex);
   395→    const joinRes = await fetch(`/api/rooms/${code}/members`, {
   396→      method: "POST",
   397→      headers: { "content-type": "application/json" },
   398→      body: JSON.stringify({
   399→        pubkey: identity.pubJwk,
   400→        ecdhPub: ecdh.pubRawB64,
   401→        alias,
   402→        colorIdx,
   403→      }),
   404→    });
   405→    if (!joinRes.ok) {
   406→      set({ sealing: null });
   407→      if (joinRes.status === 403) return { ok: false, reason: "room-full" };
   408→      return { ok: false, reason: "error" };
   409→    }
   410→    const { memberId, rejoined } = await joinRes.json();
   411→
   412→    const cipher = new RoomCipher({
   413→      roomId: code,
   414→      selfId: memberId,
   415→      sig: { privJwk: identity.privJwk, pubJwk: identity.pubJwk },
   416→      ecdh,
   417→      entryKey,
   418→    });
   419→    setCipher(code, cipher);
   420→
   421→    const session: RoomSession = {
   422→      roomId: code,
   423→      memberId,
   424→      alias,
   425→      colorIdx,
   426→      kv: 1, // the current key arrives via ECDH offers if the room has rotated
   427→      password,
   428→      creatorToken: loadCreatorToken(code),
   429→      defaultTtl: loadRoomSettings(code).defaultTtl,
   430→    };
   431→    setSession(session);
   432→    set({ sealing: null });
   433→    const name =
   434→      localName ||
   435→      get().roomCards.find((c) => c.roomId === code)?.localName ||
   436→      `Room ${code.slice(0, 4)}`;
   437→    await get().enterRoom(session, name, !!rejoined);
   438→
   439→    // The server's epoch is the rotation LEDGER (it persists; keys do
   440→    // not). If the room has rotated before and we are back at the
   441→    // password-derived key — a rejoin after refresh, or a fresh join —
   442→    // the current key must arrive via ECDH offers. Give live members a
   443→    // moment to deliver; if nobody does (everyone refreshed), the
   444→    // coordinator re-seals past the ledger so keys any departed member
   445→    // may still hold are dead again.
   446→    if (info.epoch > 1) {
   447→      const attemptRotation = (minKv: number) => {
   448→        const c = getCipher(code);
   449→        const s = getSession(code);
   450→        if (!c || !s || c.kv > 1) return;
   451→        void c.rotateTo(minKv).then((offers) => {
   452→          const relay2 = getRelay();
   453→          for (const f of offers) {
   454→            relay2.emit("message:send", { roomId: code, envelope: f });
   455→          }
   456→          addSystemLine(code, "The room re-sealed for those who remain.");
   457→        });
   458→      };
   459→      setTimeout(() => {
   460→        const cipher2 = getCipher(code);
   461→        const session2 = getSession(code);
   462→        if (!cipher2 || !session2 || cipher2.kv > 1) return;
   463→        if (cipher2.expectedCoordinator() === session2.memberId) {
   464→          attemptRotation(info.epoch);
   465→        } else {
   466→          // Ask the room for the current key; a live key-holder answers
   467→          // with an ECDH offer.
   468→          getRelay().emit("key:request", { roomId: code });
   469→          setResealing(code, true);
   470→          // Fallback: if nobody answers (every member refreshed, or the
   471→          // holder is offline), re-seal on our own after a randomized
   472→          // delay. The version jump is randomized too, so simultaneous
   473→          // fallbacks converge: a lower-version holder always accepts
   474→          // a higher-version offer.
   475→          setTimeout(() => {
   476→            const c = getCipher(code);
   477→            const s = getSession(code);
   478→            if (!c || !s || c.kv > 1) return;
   479→            void c.rotateTo(info.epoch + 1 + Math.floor(Math.random() * 8), false).then((offers) => {
   480→              const relay2 = getRelay();
   481→              for (const f of offers) {
   482→                relay2.emit("message:send", { roomId: code, envelope: f });
   483→              }
   484→              addSystemLine(code, "The room re-sealed for those who remain.");
   485→              setResealing(code, false);
   486→            });
   487→          }, 5000 + Math.floor(Math.random() * 3000));
   488→        }
   489→      }, 3000);
   490→    }
   491→    return { ok: true };
   492→  },

   516→    const relay = getRelay();

   518→      roomId: session.roomId,
   519→      memberId: session.memberId,
   520→      alias: session.alias,
   521→      colorIdx: session.colorIdx,
   522→      pubkey: cipher?.sigPubJwk,

   531→  refreshMembers: async (roomId) => {
   532→    try {
   533→      const res = await fetch(`/api/rooms/${roomId}/members`);
   534→      if (!res.ok) return;
   535→      const { members } = (await res.json()) as { members: MemberPublic[] };
   536→      set((s) => ({ members: { ...s.members, [roomId]: members } }));
   537→
   538→      // Keep the cipher's registry in sync — it is the eviction gate.
   539→      const cipher = getCipher(roomId);
   540→      if (cipher) {
   541→        const seen = new Set<string>();
   542→        for (const m of members) {
   543→          seen.add(m.memberId);
   544→          const cur = cipher.registry.get(m.memberId);
   545→          cipher.registry.set(m.memberId, {
   546→            pubkey: m.pubkey,
   547→            ecdhPubB64: m.ecdhPub ?? cur?.ecdhPubB64,
   548→            connected: cur?.connected ?? true,
   549→            joinedAt: m.joinedAt ?? cur?.joinedAt,
   550→            alias: m.alias,
   551→            colorIdx: m.colorIdx,
   552→          });
   553→        }
   554→        for (const id of [...cipher.registry.keys()]) {
   555→          if (!seen.has(id)) cipher.registry.delete(id);
   556→        }
   557→      }
   558→    } catch {
   559→      /* offline — the registry stays as-is */
   560→    }
   561→  },

   563→  /* ------------------------------------------------- send ---- */
   564→
   565→  sendMessage: async (text, file, ttlOverride) => {
   566→    const viewOnce = !!file?.viewOnce;
   567→    const { activeRoomId } = get();
   568→    if (!activeRoomId) return;
   569→    const session = getSession(activeRoomId);
   570→    const cipher = getCipher(activeRoomId);
   571→    if (!session || !cipher) return;
   572→    const trimmed = text.trim();
   573→    if (!trimmed && !file) return;
   574→
   575→    const ttlSec = (ttlOverride !== undefined ? ttlOverride : session.defaultTtl) || undefined;
   576→    const ts = Date.now();
   577→
   578→    let frames: WireFrame[];
   579→    if (file) {
   580→      const sha = await sha256HexOfBytes(fromB64(file.dataB64));
   581→      frames = await cipher.sealFile({
   582→        name: file.name,
   583→        mime: file.mime,
   584→        size: file.size,
   585→        dataB64: file.dataB64,
   586→        sha,
   587→        ttlSec,
   588→        viewOnce: viewOnce || undefined,
   589→      });
   590→    } else {
   591→      frames = await cipher.sealText({ text: trimmed, ttlSec });
   592→    }
   593→    // The first frame's id IS the message id (text frame, or the file
   594→    // meta frame) — the relay acks it and the view flips to "sent".
   595→    const id = frames[0].id;
   596→
   597→    const kind = file ? "file" : "text";
   598→    const view: MessageView = {
   599→      id,
   600→      kind,
   601→      senderId: session.memberId,
   602→      senderAlias: session.alias,
   603→      senderColor: session.colorIdx,
   604→      self: true,
   605→      status: "sending",
   606→      ts,
   607→      text: trimmed,
   608→      file: file
   609→        ? { name: file.name, mime: file.mime, size: file.size, dataB64: file.dataB64 }
   610→        : undefined,
   611→      ttlSec,
   612→      expiresAt: ttlSec ? ts + ttlSec * 1000 : undefined,
   613→      viewOnce: file && viewOnce ? true : undefined,
   614→    };
   615→
   616→    set((s) => ({
   617→      messages: {
   618→        ...s.messages,
   619→        [session.roomId]: [...(s.messages[session.roomId] ?? []), view],
   620→      },
   621→    }));
   622→    patchRoomCard(session.roomId, { lastActivity: ts });
   623→    set({ roomCards: loadRoomCards() });
   624→
   625→    if (ttlSec) {
   626→      scheduleBurn(session.roomId, id, ts + ttlSec * 1000, onMessageBurn);
   627→    }
   628→
   629→    const relay = getRelay();
   630→    for (const f of frames) {
   631→      relay.emit("message:send", { roomId: session.roomId, envelope: f });
   632→    }
   633→  },
   634→
   635→  /* --------------------------------------------- typing ------- */
   636→
   637→  emitTyping: (roomId) => {
   638→    const session = getSession(roomId);
   639→    const cipher = getCipher(roomId);
   640→    if (!session || !cipher) return;
   641→    const now = Date.now();
   642→    const last = typingLastEmit.get(roomId) ?? 0;
   643→    if (now - last < TYPING_EMIT_THROTTLE_MS) return;
   644→    typingLastEmit.set(roomId, now);
   645→    // Typing rides the same uniform encrypted frames as everything
   646→    // else — the relay cannot even tell WHEN someone is typing.
   647→    void cipher.sealTyping().then((frames) => {
   648→      const relay = getRelay();
   649→      for (const f of frames) relay.emit("message:send", { roomId, envelope: f });
   650→    });
   651→  },
   652→
   653→  /* --------------------------------------------- view-once --- */
   654→
   655→  spendViewOnce: (roomId, messageId) => {
   656→    set((s) => {
   657→      const list = s.messages[roomId] ?? [];
   658→      return {
   659→        messages: {
   660→          ...s.messages,
   661→          [roomId]: list.map((m) => (m.id === messageId ? { ...m, spent: true } : m)),
   662→        },
   663→      };
   664→    });
   665→    const cipher = getCipher(roomId);
   666→    if (cipher) {
   667→      void cipher.sealSpent(messageId).then((frames) => {
   668→        const relay = getRelay();
   669→        for (const f of frames) relay.emit("message:send", { roomId, envelope: f });
   670→      });
   671→    }
   672→  },
   673→
   674→  /* --------------------------------------------- burn-one ---- */
   675→
   676→  /** Retire one of your own letters ahead of its clock. The announce
   677→   *  is signed inside the encrypted frame, so only the author can do
   678→   *  this — nobody else can burn a message they did not write. */
   679→  burnMessage: async (roomId, messageId) => {
   680→    const { activeRoomId } = get();
   681→    const target = roomId ?? activeRoomId;
   682→    const session = getSession(target);
   683→    const cipher = getCipher(target);
   684→    const msg = (get().messages[target] ?? []).find((m) => m.id === messageId);
   685→    if (!session || !cipher || !msg || !msg.self || msg.kind === "system") return;
   686→
   687→    const frames = await cipher.sealBurn(messageId);
   688→    const relay = getRelay();
   689→    for (const f of frames) relay.emit("message:send", { roomId: target, envelope: f });
   690→
   691→    const t = burnTimers.get(messageId);
   692→    if (t) {
   693→      clearTimeout(t);
   694→      burnTimers.delete(messageId);
   695→    }
   696→    onMessageBurn(target, messageId);
   697→  },
   698→
   699→  /* ------------------------------------------------- burn ---- */
   700→
   701→  burnRoom: async (roomId) => {
   702→    const session = getSession(roomId);
   703→    if (!session?.creatorToken) return;
   704→    try {
   705→      await fetch(`/api/rooms/${roomId}/burn`, {
   706→        method: "POST",
   707→        headers: { "content-type": "application/json" },
   708→        body: JSON.stringify({ creatorToken: session.creatorToken }),
   709→      });
   710→    } catch {
   711→      /* even if this fails, members burn on the announce */
   712→    }
   713→    getRelay().emit("room:burn", { roomId });
   714→    set({ burn: { roomId } });
   715→  },
   716→
   717→  finishRoomBurn: (roomId) => {
   718→    const messages = get().messages[roomId] ?? [];
   719→    cancelBurns(roomId, messages);
   720→    dropSession(roomId);
   721→    dropCipher(roomId);
   722→    patchRoomCard(roomId, { burned: true, unread: false, lastActivity: Date.now() });
   723→    set((s) => ({
   724→      burn: null,
   725→      roomCards: loadRoomCards(),
   726→      activeRoomId: null,
   727→      messages: { ...s.messages, [roomId]: [] },
   728→    }));
   729→    get().navigate("rooms");
   730→  },
   731→
   732→  /* ------------------------------------------------- leave --- */
   733→
   734→  leaveRoom: async (roomId) => {
   735→    const session = getSession(roomId);
   736→    if (session) {
   737→      try {
   738→        await fetch(`/api/rooms/${roomId}/leave`, {
   739→          method: "POST",
   740→          headers: { "content-type": "application/json" },
   741→          body: JSON.stringify({ memberId: session.memberId }),
   742→        });
   743→      } catch {
   744→        /* best effort — receivers rotate on the announce regardless */
   745→      }
   746→      getRelay().emit("member:leave", {
   747→        roomId,
   748→        memberId: session.memberId,
   749→        alias: session.alias,
   750→      });
   751→    }
   752→    const messages = get().messages[roomId] ?? [];
   753→    cancelBurns(roomId, messages);
   754→    dropSession(roomId);
   755→    dropCipher(roomId);
   756→    removeRoomCard(roomId);
   757→    set((s) => ({
   758→      roomCards: loadRoomCards(),
   759→      activeRoomId: null,
   760→      messages: { ...s.messages, [roomId]: [] },
   761→    }));
   762→    get().navigate("rooms");
   763→  },
   764→
   765→  /* ---------------------------------------------- settings --- */
   766→
   767→  renameRoom: (roomId, name) => {
   768→    patchRoomCard(roomId, { localName: name });
   769→    set({ roomCards: loadRoomCards() });
   770→  },
   771→
   772→  setDefaultTtl: (roomId, ttl) => {
   773→    const session = getSession(roomId);
   774→    if (session) session.defaultTtl = ttl;
   775→    saveRoomSettings(roomId, { defaultTtl: ttl });
   776→  },
   777→
   778→  markVerified: (roomId, memberId, verified) => {
   779→    const current = loadVerified(roomId);
   780→    const next = verified
   781→      ? Array.from(new Set([...current, memberId]))
   782→      : current.filter((id) => id !== memberId);
   783→    saveVerified(roomId, next);
   784→    set((s) => ({ members: { ...s.members } })); // trigger re-render
   785→  },
   786→
   787→  isVerified: (roomId, memberId) => loadVerified(roomId).includes(memberId),
   788→}));

   796→  useApp.setState((s) => ({
   797→    messages: {
   798→      ...s.messages,

   801→      ),

   809→        [roomId]: (s.messages[roomId] ?? []).filter((m) => m.id !== messageId),
   810→      },
   811→    }));
   812→  }, 620);
   813→}

   856→function wireRelay(
   857→  set: (
   858→    partial:
   859→      | Partial<AppState>
   860→      | ((state: AppState) => Partial<AppState>),
   861→  ) => void,
   862→  get: () => AppState,
   863→) {
   864→  const relay = getRelay();
   865→
   866→  relay.on("connect", () => {
   867→    set({ relayOnline: true });
   868→    for (const roomId of sessionRoomIds()) {
   869→      const session = getSession(roomId);
   870→      const cipher = getCipher(roomId);
   871→      if (session && cipher) {
   872→        relay.emit("room:join", {
   873→          roomId,
   874→          memberId: session.memberId,
   875→          alias: session.alias,
   876→          colorIdx: session.colorIdx,
   877→          pubkey: cipher.sigPubJwk,
   878→          ecdhPub: cipher.ecdh.pubRawB64,
   879→          rejoined: true,
   880→        });
   881→      }
   882→    }
   883→  });
   884→
   885→  relay.on("disconnect", () => set({ relayOnline: false }));
   886→
   887→  relay.on("room:state", ({ roomId, members }: { roomId: string; members: MemberPublic[] }) => {
   888→    // merge live presence into the registry
   889→    const registry = useApp.getState().members[roomId] ?? [];
   890→    const byId = new Map(registry.map((m) => [m.memberId, { ...m }]));
   891→    for (const live of members) {
   892→      const existing = byId.get(live.memberId);
   893→      byId.set(live.memberId, {
   894→        ...live,
   895→        ...(existing?.pubkey ? { pubkey: existing.pubkey } : {}),
   896→      });
   897→    }
   898→    set((s) => ({ members: { ...s.members, [roomId]: Array.from(byId.values()) } }));
   899→
   900→    const cipher = getCipher(roomId);
   901→    if (cipher) {
   902→      for (const live of members) {
   903→        const cur = cipher.registry.get(live.memberId);
   904→        if (cur || live.pubkey) {
   905→          cipher.registry.set(live.memberId, {
   906→            pubkey: live.pubkey ?? cur?.pubkey ?? ({} as JsonWebKey),
   907→            ecdhPubB64: live.ecdhPub ?? cur?.ecdhPubB64,
   908→            connected: true,
   909→            joinedAt: cur?.joinedAt,
   910→            alias: live.alias,
   911→            colorIdx: live.colorIdx,
   912→          });
   913→        }
   914→      }
   915→    }
   916→  });
   917→
   918→  relay.on(
   919→    "member:joined",
   920→    ({ roomId, member, rejoined }: { roomId: string; member: MemberPublic; rejoined?: boolean }) => {
   921→      const session = getSession(roomId);
   922→      const cipher = getCipher(roomId);
   923→      if (!session) return;
   924→      set((s) => {
   925→        const registry = s.members[roomId] ?? [];
   926→        if (registry.some((m) => m.memberId === member.memberId)) {
   927→          return {
   928→            members: {
   929→              ...s.members,
   930→              [roomId]: registry.map((m) =>
   931→                m.memberId === member.memberId ? { ...m, connected: true } : m,
   932→              ),
   933→            },
   934→          };
   935→        }
   936→        return {
   937→          members: { ...s.members, [roomId]: [...registry, { ...member, connected: true }] },
   938→        };
   939→      });
   940→      if (cipher) {
   941→        const cur = cipher.registry.get(member.memberId);
   942→        cipher.registry.set(member.memberId, {
   943→          pubkey: member.pubkey ?? cur?.pubkey ?? ({} as JsonWebKey),
   944→          ecdhPubB64: member.ecdhPub ?? cur?.ecdhPubB64,
   945→          connected: true,
   946→          joinedAt: cur?.joinedAt,
   947→          alias: member.alias,
   948→          colorIdx: member.colorIdx,
   949→        });
   950→        // If the room has rotated past the password-derived key, hand
   951→        // the newcomer the CURRENT key — ECDH-wrapped and sealed under
   952→        // the entry key, so only a joiner who proved the password can
   953→        // open it. Every member sends; the joiner keeps the highest.
   954→        if (cipher.kv > 1 && member.memberId !== session.memberId) {
   955→          void cipher.deliverKeyTo(member.memberId).then((frames) => {
   956→            for (const f of frames) {
   957→              relay.emit("message:send", { roomId, envelope: f });
   958→            }
   959→          });
   960→        }
   961→      }
   962→      if (member.memberId !== session.memberId) {
   963→        addSystemLine(roomId, rejoined ? `${member.alias} returned.` : `${member.alias} joined.`);
   964→        touchCard(roomId, get().activeRoomId !== roomId);
   965→      }
   966→    },
   967→  );
   968→
   969→  relay.on(
   970→    "member:presence",
   971→    ({ roomId, memberId, connected }: { roomId: string; memberId: string; connected: boolean }) => {
   972→      set((s) => ({
   973→        members: {
   974→          ...s.members,
   975→          [roomId]: (s.members[roomId] ?? []).map((m) =>
   976→            m.memberId === memberId ? { ...m, connected } : m,
   977→          ),
   978→        },
   979→      }));
   980→      const cipher = getCipher(roomId);
   981→      if (cipher) {
   982→        const cur = cipher.registry.get(memberId);
   983→        if (cur) cipher.registry.set(memberId, { ...cur, connected });
   984→      }
   985→    },
   986→  );
   987→
   988→  relay.on(
   989→    "member:left",
   990→    async ({ roomId, memberId, alias }: { roomId: string; memberId: string; alias: string }) => {
   991→      const session = getSession(roomId);
   992→      const cipher = getCipher(roomId);
   993→      if (!session) return;
   994→
   995→      // Evict the leaver everywhere, first.
   996→      set((s) => ({
   997→        members: {
   998→          ...s.members,
   999→          [roomId]: (s.members[roomId] ?? []).filter((m) => m.memberId !== memberId),
  1000→        },
  1001→      }));
  1002→      if (cipher) cipher.registry.delete(memberId);
  1003→      clearTyping(roomId, memberId);
  1004→      addSystemLine(roomId, `${alias} left.`);
  1005→
  1006→      // Those who stay re-seal the room: if this client is the
  1007→      // deterministic coordinator, generate a RANDOM new key and
  1008→      // deliver it pairwise over ECDH. The leaver never receives it —
  1009→      // not through the password, not through the wire.
  1010→      if (cipher && cipher.expectedCoordinator() === session.memberId) {
  1011→        const offerFrames = await cipher.rotateAsCoordinator();
  1012→        for (const f of offerFrames) {
  1013→          relay.emit("message:send", { roomId, envelope: f });
  1014→        }
  1015→        addSystemLine(roomId, "The room re-sealed for those who remain.");
  1016→      }
  1017→
  1018→      get().refreshMembers(roomId);
  1019→    },
  1020→  );
  1021→
  1022→  relay.on("message:ack", ({ id, roomId }: { id: string; roomId?: string }) => {
  1023→    const target = roomId ?? get().activeRoomId;
  1024→    if (!target) return;
  1025→    patchView(target, id, { status: "sent" });
  1026→  });
  1027→
  1028→  // Legacy receipts from pre-upgrade clients.
  1029→  relay.on("message:spent", ({ roomId, messageId }: { roomId: string; messageId: string }) => {
  1030→    patchView(roomId, messageId, { spent: true });
  1031→  });
  1032→
  1033→  // Legacy early-burn announces from pre-upgrade clients.
  1034→  relay.on(
  1035→    "message:burn",
  1036→    async ({
  1037→      roomId,
  1038→      messageId,
  1039→      senderId,
  1040→      sig,
  1041→    }: {
  1042→      roomId: string;
  1043→      messageId: string;
  1044→      senderId?: string;
  1045→      sig?: string;
  1046→    }) => {
  1047→      const session = getSession(roomId);
  1048→      if (!session || !senderId) return;
  1049→      const target = (useApp.getState().messages[roomId] ?? []).find(
  1050→        (m) => m.id === messageId,
  1051→      );
  1052→      if (!target || target.kind === "system") return;
  1053→
  1054→      const registry = useApp.getState().members[roomId] ?? [];
  1055→      const author = registry.find((m) => m.memberId === senderId);
  1056→      const ok =
  1057→        !!author?.pubkey &&
  1058→        !!sig &&
  1059→        target.senderId === senderId &&
  1060→        (await verifyCanonical(author.pubkey, ["v1", "burn", roomId, senderId, messageId].join("|"), sig));
  1061→      if (!ok) {
  1062→        addSystemLine(
  1063→          roomId,
  1064→          `A burn request claiming to be from ${author?.alias ?? "an unknown member"} was rejected — signature invalid.`,
  1065→        );
  1066→        return;
  1067→      }
  1068→      const t = burnTimers.get(messageId);
  1069→      if (t) {
  1070→        clearTimeout(t);
  1071→        burnTimers.delete(messageId);
  1072→      }
  1073→      onMessageBurn(roomId, messageId);
  1074→    },
  1075→  );
  1076→
  1077→  relay.on("room:burned", ({ roomId }: { roomId: string }) => {
  1078→    const session = getSession(roomId);
  1079→    if (!session) return;
  1080→    set({ burn: { roomId } });
  1081→  });
  1082→
  1083→  relay.on(
  1084→    "message:new",
  1085→    async ({ roomId, envelope }: { roomId: string; envelope: WireFrame | WireEnvelope }) => {
  1086→      const session = getSession(roomId);
  1087→      if (!session || !envelope) return;
  1088→
  1089→      if ((envelope as WireFrame).v === 2) {
  1090→        await receiveFrame(roomId, envelope as WireFrame);
  1091→        return;
  1092→      }
  1093→      await receiveLegacyEnvelope(roomId, envelope as WireEnvelope);
  1094→    },
  1095→  );
  1096→
  1097→  // Legacy typing whispers from pre-upgrade clients.
  1098→  relay.on(
  1099→    "member:typing",
  1100→    ({ roomId, memberId, alias }: { roomId: string; memberId: string; alias: string }) => {
  1101→      const session = getSession(roomId);
  1102→      if (!session || memberId === session.memberId) return;
  1103→      useApp.setState((s) => {
  1104→        const list = (s.typing[roomId] ?? []).filter((t) => t.memberId !== memberId);
  1105→        return {
  1106→          typing: { ...s.typing, [roomId]: [...list, { memberId, alias, at: Date.now() }] },
  1107→        };
  1108→      });
  1109→      pruneTypingLater(roomId);
  1110→    },
  1111→  );
  1112→
  1113→  // A member who is stuck on an old key version asks for delivery.
  1114→  // Answer with the current key — wrapped so only password-holders
  1115→  // can open it. (Relay rate-limits these.)
  1116→  relay.on("key:request", ({ roomId, from }: { roomId: string; from: string }) => {
  1117→    const session = getSession(roomId);
  1118→    const cipher = getCipher(roomId);
  1119→    if (!session || !cipher || cipher.kv <= 1) return;
  1120→    if (from === session.memberId) return;
  1121→    if (!cipher.registry.has(from)) return;
  1122→    void cipher.deliverKeyTo(from).then((frames) => {
  1123→      for (const f of frames) {
  1124→        relay.emit("message:send", { roomId, envelope: f });
  1125→      }
  1126→    });
  1127→  });
  1128→}

  1130→/* ------------------------------------------------------------------
  1131→   v2 receive path — every security decision lives in the cipher
  1132→   ------------------------------------------------------------------ */
  1133→
  1134→async function receiveFrame(roomId: string, frame: WireFrame) {
  1135→  const session = getSession(roomId);
  1136→  const cipher = getCipher(roomId);
  1137→  if (!session || !cipher) return;
  1138→
  1139→  const result = await cipher.open(frame);
  1140→  await handleOpenResult(roomId, frame, result);
  1141→
  1142→  // A frame we could not yet decrypt may mean we are a joiner (or a
  1143→  // reconnecter) waiting for key delivery — ask for it, quietly.
  1144→  if (result.type === "pending" && cipher.kv === 1) {
  1145→    const now = Date.now();
  1146→    const last = keyReqLastEmit.get(roomId) ?? 0;
  1147→    if (now - last > KEYREQ_THROTTLE_MS) {
  1148→      keyReqLastEmit.set(roomId, now);
  1149→      getRelay().emit("key:request", { roomId });
  1150→      setResealing(roomId, true);
  1151→    }
  1152→  }
  1153→}
  1154→
  1155→async function handleOpenResult(roomId: string, frame: WireFrame, result: OpenResult) {
  1156→  const session = getSession(roomId);
  1157→  const cipher = getCipher(roomId);
  1158→  if (!session || !cipher) return;
  1159→  const registry = useApp.getState().members[roomId] ?? [];
  1160→  const inRoom = useApp.getState().activeRoomId === roomId;
  1161→
  1162→  switch (result.type) {
  1163→    case "text": {
  1164→      const sender = registry.find((m) => m.memberId === result.body.senderId);
  1165→      if (!sender) return;
  1166→      const ttlSec = result.body.ttlSec;
  1167→      const view: MessageView = {
  1168→        id: frame.id,
  1169→        kind: "text",
  1170→        senderId: sender.memberId,
  1171→        senderAlias: sender.alias,
  1172→        senderColor: sender.colorIdx,
  1173→        self: sender.memberId === session.memberId,
  1174→        status: "sent",
  1175→        ts: result.body.ts,
  1176→        text: result.body.text ?? "",
  1177→        ttlSec,
  1178→        expiresAt: ttlSec ? result.body.ts + ttlSec * 1000 : undefined,
  1179→      };
  1180→      useApp.setState((s) => ({
  1181→        messages: {
  1182→          ...s.messages,
  1183→          [roomId]: [...(s.messages[roomId] ?? []), view],
  1184→        },
  1185→      }));
  1186→      clearTyping(roomId, sender.memberId);
  1187→      touchCard(roomId, !inRoom);
  1188→      if (view.expiresAt) {
  1189→        scheduleBurn(roomId, view.id, view.expiresAt, onMessageBurn);
  1190→      }
  1191→      return;
  1192→    }
  1193→
  1194→    case "typing": {
  1195→      if (result.senderId === session.memberId) return;
  1196→      const sender = registry.find((m) => m.memberId === result.senderId);
  1197→      useApp.setState((s) => {
  1198→        const list = (s.typing[roomId] ?? []).filter((t) => t.memberId !== result.senderId);
  1199→        return {
  1200→          typing: {
  1201→            ...s.typing,
  1202→            [roomId]: [
  1203→              ...list,
  1204→              { memberId: result.senderId, alias: sender?.alias ?? "Someone", at: Date.now() },
  1205→            ],
  1206→          },
  1207→        };
  1208→      });
  1209→      pruneTypingLater(roomId);
  1210→      return;
  1211→    }
  1212→
  1213→    case "spent": {
  1214→      patchView(roomId, result.messageId, { spent: true });
  1215→      return;
  1216→    }
  1217→
  1218→    case "burn": {
  1219→      const target = (useApp.getState().messages[roomId] ?? []).find(
  1220→        (m) => m.id === result.messageId,
  1221→      );
  1222→      // The cipher already verified the author's signature; make sure
  1223→      // the target is actually the author's own letter.
  1224→      if (!target || target.kind === "system" || target.senderId !== result.senderId) return;
  1225→      const t = burnTimers.get(result.messageId);
  1226→      if (t) {
  1227→        clearTimeout(t);
  1228→        burnTimers.delete(result.messageId);
  1229→      }
  1230→      onMessageBurn(roomId, result.messageId);
  1231→      return;
  1232→    }
  1233→
  1234→    case "file": {
  1235→      const sender = registry.find((m) => m.memberId === result.senderId);
  1236→      if (!sender) return;
  1237→      const view: MessageView = {
  1238→        id: result.file.messageId,
  1239→        kind: "file",
  1240→        senderId: sender.memberId,
  1241→        senderAlias: sender.alias,
  1242→        senderColor: sender.colorIdx,
  1243→        self: sender.memberId === session.memberId,
  1244→        status: "sent",
  1245→        ts: result.ts,
  1246→        file: {
  1247→          name: result.file.name,
  1248→          mime: result.file.mime,
  1249→          size: result.file.size,
  1250→          dataB64: result.file.dataB64,
  1251→        },
  1252→        ttlSec: result.file.ttlSec,
  1253→        expiresAt: result.file.ttlSec ? result.ts + result.file.ttlSec * 1000 : undefined,
  1254→        viewOnce: result.file.viewOnce,
  1255→      };
  1256→      useApp.setState((s) => ({
  1257→        messages: {
  1258→          ...s.messages,
  1259→          [roomId]: [...(s.messages[roomId] ?? []), view],
  1260→        },
  1261→      }));
  1262→      clearTyping(roomId, sender.memberId);
  1263→      touchCard(roomId, !inRoom);
  1264→      if (view.expiresAt) {
  1265→        scheduleBurn(roomId, view.id, view.expiresAt, onMessageBurn);
  1266→      }
  1267→      return;
  1268→    }
  1269→
  1270→    case "file-meta":
  1271→    case "file-chunk":
  1272→      return; // assembly happens inside the cipher; render on completion
  1273→
  1274→    case "offer-installed": {
  1275→      session.kv = cipher.kv;
  1276→      setResealing(roomId, false);
  1277→      if (result.rotation) {
  1278→        addSystemLine(roomId, "The room re-sealed for those who remain.");
  1279→      }
  1280→      // Frames that were parked waiting for this key.
  1281→      const drained = await cipher.drainPending();
  1282→      for (const { frame: parked, result: r } of drained) {
  1283→        await handleOpenResult(roomId, parked, r).catch(() => undefined);
  1284→      }
  1285→      return;
  1286→    }
  1287→
  1288→    case "pending":
  1289→      return;
  1290→
  1291→    case "reject": {
  1292→      if (result.reason === "signature") {
  1293→        const sender = registry.find((m) => m.memberId === frame.from);
  1294→        addSystemLine(
  1295→          roomId,
  1296→          `A message claiming to be from ${sender?.alias ?? "an unknown member"} was rejected — signature invalid.`,
  1297→        );
  1298→        touchCard(roomId, !inRoom);
  1299→      }
  1300→      // Other rejections (replay, registry, kv, shape) are silent —
  1301→      // they are the protocol defending itself, not user-actionable.
  1302→      return;
  1303→    }
  1304→  }
  1305→}

  1308→   Legacy receive path — rooms and clients from before protocol v2

  1310→
  1311→async function receiveLegacyEnvelope(roomId: string, envelope: WireEnvelope) {
  1312→  const session = getSession(roomId);
  1313→  if (!session || !envelope) return;
  1314→
  1315→  let payload: Payload | null = null;
  1316→  try {
  1317→    const { getEpochKeyLegacy } = await import("@/lib/legacy");
  1318→    const key = await getEpochKeyLegacy(roomId, envelope.epoch, session.password);
  1319→    payload = await decryptJson<Payload>(key, envelope.iv, envelope.ct);
  1320→  } catch {
  1321→    payload = null;
  1322→  }
  1323→
  1324→  const registry = useApp.getState().members[roomId] ?? [];
  1325→  const sender = registry.find((m) => m.memberId === envelope.senderId);
  1326→
  1327→  if (!payload || !sender) {
  1328→    addSystemLine(
  1329→      roomId,
  1330→      `A message claiming to be from ${sender?.alias ?? "an unknown member"} was rejected — signature invalid.`,
  1331→    );
  1332→    touchCard(roomId, useApp.getState().activeRoomId !== roomId);
  1333→    return;
  1334→  }
  1335→
  1336→  const canonical = canonicalFor({
  1337→    roomId,
  1338→    epoch: envelope.epoch,
  1339→    senderId: envelope.senderId,
  1340→    ts: payload.ts,
  1341→    kind: envelope.kind,
  1342→    text: payload.text,
  1343→    fileSha: payload.file?.sha,
  1344→  });
  1345→  const ok = await verifyCanonical(sender.pubkey, canonical, payload.sig);
  1346→  if (!ok) {
  1347→    addSystemLine(
  1348→      roomId,
  1349→      `A message claiming to be from ${sender.alias} was rejected — signature invalid.`,
  1350→    );
  1351→    touchCard(roomId, useApp.getState().activeRoomId !== roomId);
  1352→    return;
  1353→  }
  1354→
  1355→  const view: MessageView = {
  1356→    id: envelope.id,
  1357→    kind: envelope.kind,
  1358→    senderId: sender.memberId,
  1359→    senderAlias: sender.alias,
  1360→    senderColor: sender.colorIdx,
  1361→    self: sender.memberId === session.memberId,
  1362→    status: "sent",
  1363→    ts: payload.ts,
  1364→    text: payload.text,
  1365→    file: payload.file
  1366→      ? {
  1367→          name: payload.file.name,
  1368→          mime: payload.file.mime,
  1369→          size: payload.file.size,
  1370→          dataB64: payload.file.dataB64,
  1371→        }
  1372→      : undefined,
  1373→    ttlSec: envelope.ttlSec,
  1374→    expiresAt: envelope.ttlSec ? payload.ts + envelope.ttlSec * 1000 : undefined,
  1375→    viewOnce: envelope.viewOnce,
  1376→  };
  1377→  useApp.setState((s) => ({
  1378→    messages: {
  1379→      ...s.messages,
  1380→      [roomId]: [...(s.messages[roomId] ?? []), view],
  1381→    },
  1382→  }));
  1383→  clearTyping(roomId, sender.memberId);
  1384→  touchCard(roomId, useApp.getState().activeRoomId !== roomId);
  1385→  if (view.expiresAt) {
  1386→    scheduleBurn(roomId, view.id, view.expiresAt, onMessageBurn);
  1387→  }
  1388→}
